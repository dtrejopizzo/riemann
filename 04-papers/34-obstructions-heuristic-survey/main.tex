\title[Obstructions in a series of approaches to RH]{Obstructions in a series of approaches to the Riemann Hypothesis:\\ a heuristic survey}

\author{David Alejandro Trejo Pizzo}
\thanks{Independent Researcher. \texttt{dtrejopizzo@gmail.com}}


\begin{document}

\begin{abstract}
We survey the approaches to the Riemann Hypothesis (RH) studied in Papers~02--21 of
this series: logarithmic sums of $q^{\om(n)}$ and their shifted correlations, localized
Weil quadratic forms, Kre\u\i n and Pontryagin formulations, de Branges spaces, and
Hodge-index, hyperbolicity and Lefschetz analogies. For each we state what the
corresponding paper proves, with its hypotheses, and describe heuristically where the
approach stops. The proved statements recalled include: the Selberg--Delange asymptotics
of $M_q(N)=\sum_{n\le N}q^{\om(n)}/n$ and the bound
$C(h,N;q)/M_q(N)^2\ll1/\log N$ for the shifted correlations when $q\ge1$ (Paper~21); the bound
$\kV\le m$ for the negative index of the Weil form on even tests when there are
$m<\infty$ off-line quartets (Paper~18); the second-moment identity for zero exponential
sums (Paper~16); the monotonicity of the off-line mass of a real polynomial with simple zeros under the
backward heat equation, the polynomial model of the de Bruijn--Newman flow (Paper~13); the moment identity for the Taylor coefficients of $\Xi$ and an upper bound
for the least eigenvalue of a sine-kernel Gram matrix (Paper~14); and the results on the
archimedean term and on $\mathfrak{sl}_2$ triples of Paper~17. Several further statements
in the series are numerical observations, which we label as such. The organisation of
the stopping points into three ``walls'' and the principle that sign data is not
supplied by symmetry data are heuristic. Nothing here bears on the truth of RH.

\medskip
\noindent\textbf{Key words and phrases.}
Riemann Hypothesis, explicit formula, Weil positivity, negative index, Pontryagin spaces,
de Branges spaces, $\om(n)$, Selberg--Delange method, survey.
\end{abstract}

\subsection{The Riemann Hypothesis and its equivalent forms}
\label{ss:RH}

Let $\zeta(s) = \sum_{n=1}^\infty n^{-s}$ for $\Re(s)>1$, extended meromorphically to
$\mathbb{C}$.  Its non-trivial zeros $\rho = \sig_\rho + i\gam_\rho$ satisfy
$0 < \sig_\rho < 1$ and are symmetric under $\rho\mapsto\bar\rho$ and
$\rho\mapsto1-\bar\rho$.

\begin{definition}[The Riemann Hypothesis]
$\RH$: every non-trivial zero $\rho$ of $\zeta(s)$ satisfies $\sig_\rho = \half$.
\end{definition}

\begin{proposition}[Equivalent forms of RH used below]
\label{prop:RH-equiv}
The following are equivalent:
\begin{enumerate}[label=\upshape(\roman*)]
  \item \textup{(RH)} Every non-trivial zero of $\zeta$ has real part $\half$.
  \item \textup{(PNT-error)} For all $\eps>0$,
    $\psi(x) = x + O(x^{1/2+\eps})$, where $\psi(x)=\sum_{n\le x}\Lam(n)$.
  \item \textup{(Weil positivity)} The Weil quadratic form
    \eqref{eq:weil-form} satisfies $B(f,f)\ge0$ for all $f\in C_c^\infty(\mathbb R)$.
  \item \textup{(Negative index)} The negative index $\kap$ of the Weil form on
    $C_c^\infty(\mathbb R)$ is zero (equivalent to \textup{(iii)} by the definition of
    the negative index).
\end{enumerate}
\end{proposition}

\begin{proof}
(i)$\Leftrightarrow$(ii) is von Koch's theorem~\cite{vK01}; see also
Paper~21~\cite[Thm.~6.1]{P21}.
(i)$\Leftrightarrow$(iii) is Weil's criterion~\cite{Weil52,Bom00}; see \S\ref{ss:Weil}.
(iii)$\Leftrightarrow$(iv): a form has negative index $0$ if and only if it is
non-negative on the space considered (Definition~\ref{def:index}).
\end{proof}

\subsection{The explicit formula}
\label{ss:explicit}

For $f\in C_c^\infty((1,\infty))$ let
$F(s)=\int_0^\infty f(x)x^{s-1}\,dx$ be its Mellin transform, an entire function.

\begin{theorem}[Explicit formula; classical]
\label{thm:explicit}
For $f\in C_c^\infty((1,\infty))$,
\begin{equation}
  \label{eq:explicit}
  \sum_{n=1}^\infty \Lam(n)f(n)
  = F(1) - \sum_\rho F(\rho) - \sum_{k=1}^\infty F(-2k),
\end{equation}
where $\rho$ runs over the non-trivial zeros of $\zeta$ with multiplicity, the sums
converging absolutely.
\end{theorem}

This follows by shifting the contour in
$\frac1{2\pi i}\int_{(2)}(-\zeta'/\zeta)(s)F(s)\,ds$ to the left: the residues are $F(1)$
at $s=1$, $-F(\rho)$ at the non-trivial zeros and $-F(-2k)$ at the trivial zeros; there is
no residue at $s=0$, since $F$ is entire and $\zeta(0)\ne0$. In von Mangoldt's formula
$\psi_0(x)=x-\sum_\rho x^\rho/\rho-\log2\pi-\half\log(1-x^{-2})$, which corresponds
formally to $F(s)=x^s/s$, the constant $-\log2\pi=-\zeta'(0)/\zeta(0)$ is the residue at the
pole of $x^s/s$ at $s=0$.

\subsection{The Selberg--Delange method}
\label{ss:SD}

\begin{theorem}[Selberg--Delange, Tenenbaum \cite{Ten15}, Thm.~II.5.2]
\label{thm:SD}
Let $z\in\mathbb{C}$ and
\[
F(s) = \sum_{n\ge1} a(n)n^{-s} = \zeta(s)^z G(s),
\]
where $G(s)=\sum_{n\ge1}b(n)n^{-s}$ satisfies $\sum_n|b(n)|n^{-\sigma_0}<\infty$ for some
$\sigma_0<1$, and the coefficients $a(n)$ satisfy the growth hypotheses of the cited
theorem (they hold, for instance, for $a(n)=q^{\om(n)}$ with $q>0$). Then for every $K\ge0$,
\[
  \sum_{n\le x} a(n) =
  x (\log x)^{z-1}
  \Bigl\{ \sum_{k=0}^K \frac{\lambda_k}{(\log x)^k} + O\Bigl(\frac{1}{(\log x)^{K+1}}\Bigr)\Bigr\},
  \qquad \lambda_0=\frac{G(1)}{\Gamma(z)},
\]
where the $\lambda_k$ are determined by the Taylor coefficients at $s=1$ of
$s^{-1}G(s)\bigl((s-1)\zeta(s)\bigr)^z$, with $\lambda_k$ carrying the factor
$1/\Gamma(z-k)$.
\end{theorem}

\begin{remark}
The contour used in the proof lies in the classical zero-free region of $\zeta$, so the
expansion contains no term indexed by the zeros. For non-integer $z$ the function $\zeta^z$
has branch points at the zeros of $\zeta$, and the method avoids them.
\end{remark}

\subsection{Notation summary}
\label{ss:notation}

Throughout the paper:
\begin{itemize}
\item $\om(n)$ = number of \emph{distinct} prime factors of $n$.
\item $q>0$ is a real parameter; the case $q=k^2$ relates to the moments of $\zeta$.
\item $F_q(s) = \sum_{n\ge 1} q^{\om(n)} n^{-s} = \zeta(s)^q G_q(s)$, with
  $G_q$ as in Proposition~\ref{prop:factorization}.
\item $M_q(N) = \sum_{n\le N} q^{\om(n)}/n$ and
  $C(h,N;q) = \sum_{n\le N} q^{\om(n)}q^{\om(n+h)}/n$, as in Paper~21~\cite{P21}.
\item $\kap$ = negative index of a quadratic form (Definition~\ref{def:index});
  $\kV$ = negative index of the Weil form on even real tests (Paper~18~\cite{P18}).
\item $\sig_0 = \sup\{\Re\rho : \zeta(\rho)=0,\; \rho \text{ non-trivial}\}$.
  RH says $\sig_0 = \half$.
\end{itemize}

\subsection{Motivation: moments of $\zeta$ on the critical line}
\label{ss:omega-motivation}

The Keating--Snaith conjecture~\cite{KS00} predicts that
\[
  \frac{1}{T}\int_0^T |\zeta(\half + it)|^{2k} dt \sim c_k (\log T)^{k^2}
\]
for an explicit constant $c_k$. Since $d_k(n)^2=(k^2)^{\om(n)}$ on squarefree $n$, the
exponent $k^2$ is the Selberg--Delange exponent of the diagonal sum
$\sum_{n\le N}d_k(n)^2/n$ (classical; Paper~05~\cite[Thm.~3.2]{P05}). Papers~05 and~21
study the weights $q^{\om(n)}$ from this point of view.

\subsection{The Dirichlet series of $q^{\omega(n)}$}
\label{ss:dirichlet}

\begin{proposition}[Factorization]
\label{prop:factorization}
For $q > 0$ and $\Re(s) > 1$:
\[
  F_q(s) = \sum_{n=1}^\infty \frac{q^{\om(n)}}{n^s}
  = \prod_p \frac{1+(q-1)p^{-s}}{1-p^{-s}}
  = \zeta(s)^q \cdot G_q(s),
  \qquad
  G_q(s) = \prod_p \bigl(1+(q-1)p^{-s}\bigr)\bigl(1-p^{-s}\bigr)^{q-1}.
\]
\end{proposition}

\begin{proof}
At each prime $p$, $\om(p^k) = 1$ for all $k\ge 1$, so the local Euler factor is
\[
  \sum_{k=0}^\infty \frac{q^{\om(p^k)}}{p^{ks}}
  = 1 + \frac{q}{p^s-1}
  = \frac{1+(q-1)p^{-s}}{1-p^{-s}}.
\]
Since $\zeta(s)^q = \prod_p (1-p^{-s})^{-q}$, we get
$G_q(s)=F_q(s)\zeta(s)^{-q}=\prod_p(1+(q-1)p^{-s})(1-p^{-s})^{q-1}$.
\end{proof}

\begin{lemma}[holomorphy of $G_q$]
\label{lem:Gq}
For $q>0$, $G_q$ is holomorphic in $\Re s>\half$; it is non-vanishing there if and only
if $q\le1+\sqrt2$ \textup{(}Paper~21~\cite[Lemma~2.1]{P21}\textup{)}.
\end{lemma}

\begin{proof}
The function $\phi(x)=(1+(q-1)x)(1-x)^{q-1}$ is holomorphic in $|x|<1$ with $\phi(0)=1$
and $\phi'(0)=(q-1)-(q-1)=0$, so $|\phi(x)-1|\le C_q|x|^2$ for $|x|\le2^{-1/2}$. With
$x=p^{-s}$ and $\sigma=\Re s>\half$ one has $|x|=p^{-\sigma}<2^{-1/2}$, and
$\sum_pp^{-2\sigma}$ converges locally uniformly, so the product converges absolutely and
locally uniformly and $G_q$ is holomorphic. Its zeros are those of the factors; $(1-x)^{q-1}$
does not vanish, and $1+(q-1)p^{-s}=0$ requires $q\ne1$ and $p^{-\sigma}=1/|q-1|$. For
$0<q<1$ this is impossible since $1/(1-q)>1$. For $q>1$ it is possible with $\sigma>\half$
if and only if $(q-1)^{-1}<p^{-1/2}$ for some $p$, i.e.\ $q-1>\sqrt2$ (take $p=2$ and
$2^{-s}=-1/(q-1)$).
\end{proof}

\subsection{Asymptotics of $M_q(N)$}
\label{ss:zerosofzeros}

\begin{proposition}
\label{thm:Mk-monotone}
For $q>0$, $M_q(N)$ is strictly increasing in $N$ and
\[
  M_q(N)=C_q(\log N)^q+O_q\bigl((\log N)^{q-1}+1\bigr),\qquad C_q=\frac{G_q(1)}{\Gamma(q+1)};
\]
for integer $q\ge1$, $M_q(N)=P_q(\log N)+O_{q,\delta}(N^{-\delta})$ for every
$\delta<\min(\half,1/q)$, where $P_q$ is a polynomial of degree $q$ with leading coefficient
$G_q(1)/q!$ \textup{(}Paper~21~\cite[Prop.~3.1 and Thm.~9.1]{P21}\textup{)}.
\end{proposition}

Monotonicity holds because every term $q^{\om(n)}/n$ is positive. The constant is
$G_q(1)/\Gamma(q+1)$ rather than $G_q(1)/\Gamma(q)$ because $M_q$ sums $a(n)/n$, whose
Dirichlet series is $F_q(s+1)$. The Selberg--Delange contour stays in a zero-free region of
$\zeta$, so no term indexed by zeros appears in the expansion.

\begin{remark}[No oscillation]
For integer $q$ the error term $O(N^{-\delta})$ holds unconditionally
(Paper~21~\cite[Cor.~6.2]{P21}), so no bound of this form for $M_q(N)$ distinguishes
whether RH holds: $M_q$ is determined, up to this error, by the singularity of $F_q(s+1)$
at $s=0$.
\end{remark}

\subsection{The exponent and large deviations of $\om(n)$}
\label{ss:k2}

\begin{proposition}
\label{thm:k2}
For $q>0$, $\lim_{N\to\infty}\log M_q(N)/\log\log N=q$. Under the counting measure on
$n\le N$, $\om(n)/\log\log N$ satisfies a large-deviation principle with speed
$\log\log N$ and rate $I(\alpha)=\alpha\log\alpha-\alpha+1$ \textup{(}classical; see
Paper~21~\cite[Thm.~4.1]{P21}\textup{)}.
\end{proposition}

\begin{proof}
By Proposition~\ref{thm:Mk-monotone}, $\log M_q(N)=q\log\log N+\log C_q+o(1)$, since
$G_q(1)>0$. The large-deviation principle follows from the Sathe--Selberg asymptotics by
the G\"artner--Ellis theorem, as in Paper~21.
\end{proof}

The exponent $q=k^2$ is the Selberg--Delange exponent of $d_k(n)^2$ on squarefree $n$;
this is classical (Paper~05~\cite[Thm.~3.2]{P05}).

\subsection{Shifted correlations}
\label{ss:correlation}

\begin{definition}[Shifted correlation]
For $h \ge 1$ and $q > 0$ put
\begin{align}
  C(h,N;q) &= \sum_{n\le N}\frac{q^{\om(n)}\,q^{\om(n+h)}}{n}, \\
  q^{\om(n)} &= \sum_{d\mid n,\ d\ \mathrm{sqfree}}(q-1)^{\om(d)} .
\end{align}
The second line is the identity used to expand $C(h,N;q)$.
\end{definition}

\begin{proposition}[{Paper~21~\cite[Prop.~5.2 and Thm.~5.3]{P21}}]
\label{thm:T4}
For $q\ge1$ and $h\ge1$: $\log N\le C(h,N;q)\ll_{q,h}(\log N)^{2q-1}$, hence
$C(h,N;q)/M_q(N)^2\ll1/\log N$. Moreover, for $q>0$,
\begin{equation}
  \label{eq:Aq}
  C(h,N;q)=\sum_{\substack{d,e\ \mathrm{sqfree}\\\gcd(d,e)\mid h}}(q-1)^{\om(d)+\om(e)}
  \sum_{\substack{n\le N\\ d\mid n,\ e\mid n+h}}\frac1n ,
\end{equation}
and each inner sum equals
$\frac{\log N}{m}-\frac{\psi(r/m)+\log m}{m}+O(1/N)$, where $m=\operatorname{lcm}(d,e)$,
$r$, $1\le r\le m$, represents the residue class defined by the two conditions, and $\psi$ is the digamma
function.
\end{proposition}

Each inner sum is a harmonic sum over one progression; no zero of $\zeta$ or of a
Dirichlet $L$-function enters.

\begin{remark}
\label{thm:T4.1}
No equivalent of RH in terms of $M_q$ or $C(h,N;q)$ is known (Paper~21, after
Theorem~6.1).
\end{remark}

\begin{proposition}[{Paper~21~\cite[Thm.~7.1]{P21}}]
\label{thm:T5}
For $q_1,\dots,q_r>0$ and integers $h_1,\dots,h_r\ge0$, the local density
\[
  \delta_p=\frac1p\sum_{a\bmod p}\prod_{j}q_j^{\mathbf 1[p\mid a+h_j]}
  =1+\frac1p\sum_{b\in B_p}\Bigl(\prod_{j:\,h_j\equiv b\ (p)}q_j-1\Bigr)
\]
is positive for every $p$, where $B_p$ is the set of residues of $h_1,\dots,h_r$
modulo $p$.
\end{proposition}

\begin{proof}
$\delta_p$ is a mean of products of positive numbers.
\end{proof}

\begin{remark}
Proposition~\ref{thm:T5} concerns the multiplicative local densities of the correlation
sums $\sum_{n\le N}n^{-1}\prod_jq_j^{\om(n+h_j)}$; it has no consequence for the zeros
of $\zeta$.
\end{remark}

\subsection{The Weil explicit formula as a quadratic form}
\label{ss:Weil}

We use the normalisation of Paper~07~\cite{P07}.

\begin{definition}[Weil quadratic form]
Let $h$ be holomorphic in $|\Im r|\le\half+\eps$ with $h(r)\ll(1+|r|)^{-2-\eps}$ there,
and $g(u)=\frac1{2\pi}\int h(r)e^{-iru}\,dr$. The Riemann--Weil explicit formula reads
\[
  \sum_\rho h(\gam_\rho)=h(\tfrac i2)+h(-\tfrac i2)+\frac1{2\pi}\int_{-\infty}^\infty
  h(r)\,\Omega(r)\,dr-\sum_{n\ge2}\frac{\Lam(n)}{\sqrt n}\bigl(g(\log n)+g(-\log n)\bigr),
\]
with $\rho=\half+i\gam_\rho$ and
$\Omega(r)=\Re\frac{\Gamma'}{\Gamma}(\tfrac14+\tfrac{ir}2)-\log\pi$. For $f\in
C_c^\infty(\mathbb R)$ put $\hat f(r)=\int f(u)e^{iru}\,du$, so that
$h(r)=\hat f(r)\overline{\hat f(\bar r)}$ is admissible; the Weil form is
\begin{equation}
  \label{eq:weil-form}
  B(f,f)=\sum_\rho \hat f(\gam_\rho)\overline{\hat f(\bar\gam_\rho)} .
\end{equation}
\end{definition}

\begin{theorem}[Weil's criterion \cite{Weil52}; see \cite{Bom00}]
$\RH \Longleftrightarrow B(f,f) \ge 0$ for all $f\in C_c^\infty(\mathbb R)$.
\end{theorem}

Paper~07 studies the finite sections $Q=M_{\mathrm{zeros}}-M_{\mathrm{arith}}$ of this
identity on a Hermite--Gauss basis localized at a height $T_0$; this localization is a
restriction of the test class, not a separate criterion.

\subsection{The negative index}
\label{ss:Krein}

\begin{definition}[Negative index]
\label{def:index}
The \emph{negative index} of a Hermitian form $Q$ on a vector space $\mathcal{H}$ is
\[
  \kap(Q) = \sup\{\dim V : V\subset\mathcal{H},\; Q(v,v)<0\ \text{for}\ 0\ne v\in V\}.
\]
\end{definition}

\begin{proposition}[{Paper~18~\cite[Cor.~6.5 and Prop.~7.2]{P18}}]
\label{thm:kappa}
If $\zeta$ has exactly $m$ off-line quartets of zeros, $1\le m<\infty$, the negative index
$\kV$ of the Weil form on even real tests satisfies $\kV\le m$; equality is open. By the
definition of the negative index and Weil's criterion, RH implies $\kV=0$.
\end{proposition}

Paper~08~\cite{P08} writes the zero side of the Weil form on Paley--Wiener spaces as
$\langle JEF,EF\rangle$, with $J=I$ exactly under RH; the representation by a bounded
self-adjoint operator holds under a coercivity hypothesis that is not proved there.

\begin{proposition}[{Paper~07~\cite[Lemma~3.1 and Thm.~3.2]{P07}}]
\label{thm:localization}
Let $J\ge2$ be the number of Hermite--Gauss basis functions. Replacing a double point
$\half+i\gam$ \textup{(}$\gam$ real\textup{)} of a configuration by the two points
$\half\pm\delta+i\gam$ changes the zero-side Gram matrix on the localized basis by
$P(\gam,\delta)=\delta^2A(\gam)+R$, $\|R\|\le C(\gam,\delta)\delta^4$, with $A(\gam)$ of
rank $\le3$ and $\lambda_{\min}(A(\gam))<0$; for the quartet $\half\pm\delta\pm i\gam$ the
change is $P(\gam,\delta)+P(-\gam,\delta)$.
\end{proposition}

\begin{nogobox}{Numerical observation (Paper~12)}
Paper~12~\cite[Obs.~3.1 and 4.1]{P12}: at $T_0=1000$ and $2d/\log T_0\in\{0.87,1.45\}$
the largest generalized eigenvalue of the prime Gram matrix relative to the archimedean
one equals $1$ to five digits, and at $T_0=80$, $d=1.3$ the diagonal-normalised zero Gram
matrix agrees with the sine kernel to $7\times10^{-2}$. No theorem is claimed.
\end{nogobox}

\subsection{Detection versus proof}

For the model configuration of Paper~07 (a formal signed multiset: the zeros of
$\zeta$, the four points $\half\pm\delta\pm i\gam$ with multiplicity $+1$ and the points
$\half\pm i\gam$ with multiplicity $-2$, used with the arithmetic side of $\zeta$), under
the smallness conditions of \cite[Thm.~5.1]{P07} the truncated form $Q_X$ has a negative
eigenvalue for all cutoffs $X\ge X_{\min}$, with $X_{\min}$ explicit
(Paper~07). This shows that the finite sections react to an
off-line quartet; it gives no information on whether $\zeta$ has one.

\begin{wallbox}{the Weil-form direction}
The finite sections of the Weil form detect a model off-line quartet. Proving that the
negative index vanishes is Weil's criterion, i.e.\ RH itself. Paper~18 bounds $\kV$ by
the number of off-line quartets when that number is finite, and no unconditional bound on
$\kV$ is known.
\end{wallbox}

\subsection{A heuristic organising principle}

\begin{remark}[Heuristic: the sign mismatch]
\label{thm:capstone}
Many unconditional estimates take the form of lower bounds for averaged quantities,
while RH asks that a negative index vanish. Paper~14~\cite{P14} argues heuristically that
the Hodge-index and Jensen-hyperbolicity routes reduce to positivity statements of Weil
type. This is a heuristic organising principle, not a theorem.
\end{remark}

\subsection{Comparison of positivity mechanisms}

$G$ is the sine-kernel Gram matrix of $K$ consecutive zero ordinates normalised to mean
spacing one and $\beta_{\min}$ their least gap; $b(k)=(-1)^k\Xi^{(2k)}(0)/(2k)!$ and
$m_{2k}=\int_0^\infty u^{2k}\Phi(u)\,du$; $z_j=x_j+iy_j$ are the zeros of the polynomial and
$L=\sum y_j^2$; $\mathcal H_d$ is the space of $L^2$ functions supported in $[-d,d]$ and
$\kap_d$ the negative index of the Weil form on it.

\begin{table}[h]
\small

\subsection{The second-moment identity}

Following Paper~16, fix $T$ and $H\ge1$; let $\gam=\Im\rho$ run over the zeros $\rho$ with
$T<\Im\rho\le T+H$, counted with multiplicity, and let $N$ be their number. Put
$S(\xi)=\sum_\gam e^{i\gam\xi}$ and
$V(d,T)=\sum_{\gam,\gam'}2\sin(2d(\gam-\gam'))/(\gam-\gam')$, the summand read as $4d$ for
$\gam=\gam'$; put
$V_w(d,T)=\sum_{0<\gam,\gam'\le T}w(\gam-\gam')\,2\sin(2d(\gam-\gam'))/(\gam-\gam')$ with
Montgomery's weight $w(u)=4/(4+u^2)$; $N(T)$ is the number of zeros with $0<\gam\le T$, and
$L=\log T$.

\begin{proposition}[{Paper~16~\cite[Thm.~2.1, Prop.~4.1, Cor.~5.2]{P16}}]
\label{thm:LB}
$V(d,T)=\int_{-2d}^{2d}|S(\xi)|^2d\xi$, and $0\le V(d,T)\le4dN^2$.
Assuming RH, for fixed $A\ge1$ and $d=AL/2$, $V_w(d,T)\ll dN(T)$; assuming in addition
the uniform form-factor condition of Goldston--Montgomery, $V_w(d,T)=(4+o(1))dN(T)$.
\end{proposition}

\begin{theorem}[Goldston--Montgomery \cite{GM87}; Paper~16, Thm.~5.1]
Assume RH. The asymptotic $F(\alpha,T)\sim1$ uniformly for $1\le\alpha\le M$, for every
fixed $M$, is equivalent to the asymptotic
$\int_1^X(\psi(x+\delta x)-\psi(x)-\delta x)^2dx\sim\half\delta X^2\log(1/\delta)$
uniformly for $X^{-1}\le\delta\le X^{-\eps}$, for every fixed $\eps>0$.
\end{theorem}

Here $F(\alpha,T)$ is Montgomery's pair-correlation function~\cite{Mon73}.

\begin{remark}
Paper~11~\cite[Prop.~2.1 and Rem.~2.2]{P11} proves $\kap_d<\infty$ on each $\mathcal H_d$; uniformity
in $d$ is open. No unconditional bound of the form ``spectral bottom $\ge-C(\log T)^c$''
for the full Weil form is proved in the series.
\end{remark}

\begin{nogobox}{Heuristic summary}
The proved statements of this section are unconditional identities and bounds
(Table~\ref{tab:capstone}) or statements under RH (Proposition~\ref{thm:LB}). None of them
bounds the negative index of the Weil form.
\end{nogobox}

\section{Geometric Analogies: Hodge Index and Lefschetz}
\label{sec:geometric}
%% =====================================================================

\subsection{The function-field model}

\begin{remark}[Heuristic]
In the function-field case the positivity comes from the Hodge index theorem on $C\times C$
(Weil~\cite{Weil48}); no analogous surface over $\Spec\mathbb Z$ is known, and its
construction is the aim of programmes such as that of Connes and Consani.
\end{remark}

\begin{proposition}[{Paper~17~\cite[Props.~3.1, 4.1, 5.1, 7.1]{P17}}]
\label{thm:lefschetz}
\begin{enumerate}[label=\upshape(\roman*)]
\item $\Psi(r)=\Re\frac{\Gamma'}{\Gamma}(\tfrac14+\tfrac{ir}2)-\log\pi$ \textup{(}the function $\Omega$ of \S\ref{ss:Weil}, denoted $\Psi$ in Paper~17\textup{)} is strictly increasing on
$r>0$ with unique positive zero $r_0=6.2898\ldots$.
\item The archimedean term is negative on $\widehat f_\sigma(r)=(r^2+\frac14)e^{-r^2/2\sigma^2}$
for small $\sigma$ and positive for large $\sigma$.
\item The standard $\mathfrak{sl}_2$ triple of Paper~17 on $L^2(\mathbb R)$ has no
lowest-weight vector.
\item An $\mathfrak{sl}_2$-string of length $\ge3$ graded by the ordinates would give a
three-term arithmetic progression among them, excluded under the linear independence
hypothesis (LI).
\end{enumerate}
\end{proposition}

\begin{remark}[Heuristic]
A Hilbert--P\'olya strategy (a self-adjoint operator with spectrum $\{\gam_\rho\}$, a
positivity statement, and the conclusion $\gam_\rho\in\mathbb R$) requires an operator
that is not known; Proposition~\ref{thm:lefschetz} shows only that two specific
$\mathfrak{sl}_2$ structures do not provide one.
\end{remark}

\begin{wallbox}{the geometric direction}
No arithmetic surface playing the role of $C\times C$ is known; without it the Hodge-index
argument has no counterpart for $\zeta$.
\end{wallbox}

\subsection{Eventual versus uniform positivity}

\begin{remark}
\label{thm:finitization}
Paper~15~\cite{P15} separates eventual and uniform positivity in scale-dependent
hierarchies, records when one implies the other~\cite[Thm.~3.1]{P15}, and proves a
continuity lemma for the Hodge--Riemann gap of a convergent family of forms~\cite[Lemma~8.2]{P15}.
It proves no statement about $\zeta$.
\end{remark}

\begin{remark}
Over a finite field $\mathbb F_\ell$ the Frobenius has infinite order, and its eigenvalues on
$H^1$ have absolute value $\ell^{1/2}$; the finiteness that the function-field proof uses
is the finite-dimensionality of $H^1$, which has no known analogue for $\zeta$.
\end{remark}

\section{Kre\u\i n and Pontryagin Spaces and de Branges Spaces}
\label{sec:functional}
%% =====================================================================

\subsection{Kre\u\i n and Pontryagin spaces}

\begin{definition}[Kre\u\i n and Pontryagin spaces]
A \emph{Kre\u\i n space} is a Hilbert space $(\mathcal{H}, \langle\cdot,\cdot\rangle)$
equipped with a fundamental symmetry $J$ ($J^2=I$, $J^*=J$) and the inner product
$[x,y] = \langle Jx,y\rangle$.  A \emph{Pontryagin space} is a Kre\u\i n space with
$\kap(J):=\dim\ker(J+I)<\infty$.
\end{definition}

\begin{theorem}[Pontryagin; see Iokhvidov--Kre\u\i n--Langer \cite{IKL82}]
\label{thm:KL}
A Pontryagin space $\Pi_\kap$ admits fundamental decompositions
$\mathcal H_+[\dotplus]\mathcal H_-$ with $\dim\mathcal H_-=\kap$. A self-adjoint operator
in $\Pi_\kap$ has at most $\kap$ eigenvalues \textup{(}with algebraic multiplicity\textup{)}
in $\{\Im z>0\}$, and its non-real spectrum is symmetric about $\mathbb R$.
\end{theorem}

\begin{nogobox}{Paper~11}
Paper~11~\cite[Prop.~2.1]{P11} proves a conditional reduction: if the archimedean form has
compact resolvent and the prime tail beyond $X$ is relatively form-bounded with bound
$<1$, then $\kap<\infty$; on $\mathcal H_d$ the hypotheses hold, so $\kap_d<\infty$
(\cite[Rem.~2.2]{P11});
uniformity in $d$ is open. Exploratory computations, not reproduced there, suggest that
the uniform form of the hypothesis fails.
\end{nogobox}

\subsection{De Branges spaces and axiomatics}

\begin{definition}[De Branges space]
An entire function $E(z)$ satisfying $|E(\bar z)| < |E(z)|$ for $\Im z > 0$
defines a de~Branges space $H(E) = \{f \text{ entire}: f/E,\ f^\#/E\in H^2(\mathbb{C}^+)\}$
with inner product $\langle f,g\rangle = \int_{-\infty}^\infty \overline{f(t)}g(t)/|E(t)|^2\,dt$.
\end{definition}

\begin{theorem}[de Branges \cite{dB68}, Thm.~23]
A Hilbert space $\mathcal{H}\ne\{0\}$ of entire functions equals $H(E)$ isometrically for
some $E$ if and only if:
\begin{itemize}
  \item[\textup{(dB1)}] for every non-real $w$, $f\mapsto f(w)$ is bounded;
  \item[\textup{(dB2)}] if $f\in\mathcal H$, $w\notin\mathbb R$ and $f(w)=0$, then
    $f(z)(z-\bar w)/(z-w)\in\mathcal H$ and has the same norm;
  \item[\textup{(dB3)}] $f\in\mathcal{H} \Rightarrow
    f^\#(z):=\overline{f(\bar z)}\in\mathcal{H}$ with $\|f^\#\|=\|f\|$.
\end{itemize}
\end{theorem}

\begin{proposition}[{Paper~18~\cite[Thms.~8.2, 8.4, Prop.~8.3, Rems.~6.6, 8.5]{P18}}]
\label{thm:rigidity}
A realization of the Weil form in a Pontryagin space of entire functions satisfying
\textup{(dB1)--(dB3)} is a space $H(E)$ with negative index $\ge\kV$
\textup{(}Kaltenb\"ack--Woracek~\cite{KW99}\textup{)}; the conjugation axiom holds under a
compatibility condition. The function $E_\xi(z)=\xi(\half+iz)$ satisfies
$E_\xi^\#=E_\xi$, so $H(E_\xi)=\{0\}$ and $E_\xi$ is not a structure function.
\end{proposition}

\subsection{Symmetry versus positivity}
\label{ss:sym-pos}

\begin{proposition}[{Paper~18~\cite[Thm.~9.5]{P18}}]
\label{thm:sym-pos}
The zero set of $\xi$ is invariant, with multiplicities, under the Klein four-group
generated by $s\mapsto\bar s$ and $s\mapsto1-s$; the anti-holomorphic involution
$j(s)=1-\bar s$ has the critical line as fixed set. RH is the pointwise condition
$\{\rho\}\subset\operatorname{Fix}(j)$, which is not implied by the setwise invariance.
\end{proposition}

\begin{remark}[Heuristic]
The symmetry $\rho\mapsto1-\bar\rho$ acts on every zero configuration, on- or off-line,
so it cannot by itself force $\kap=0$.
\end{remark}

\begin{wallbox}{the functional-analytic direction}
The symmetry data of the zeros holds unconditionally, while the vanishing of the negative
index is RH. Montgomery's pair-correlation theorem (under RH) and the GUE conjectures
describe the statistics of the zeros; no implication from them to RH is known.
\end{wallbox}

\section{Remarks on where the Approaches Stop}
\label{sec:nogo}
%% =====================================================================

We collect the remarks above. None of them is a theorem excluding a method.

\begin{nogobox}{Sign mismatch (heuristic)}
Remark~\ref{thm:capstone}: unconditional estimates are typically lower bounds for
averages, while RH asks for the vanishing of a negative index.
\end{nogobox}

\begin{nogobox}{Saturation and sine-kernel structure (numerical)}
Paper~12~\cite[Obs.~3.1, 4.1]{P12}: at $T_0=1000$ on a $29$-dimensional sinc basis the
largest generalized eigenvalue of the prime Gram matrix relative to the archimedean one
equals $1$ to five digits at $2d/\log T_0\in\{0.87,1.45\}$ (and is below $1$ at smaller
$d$); at $T_0=80$, $d=1.3$ the diagonal-normalised zero Gram matrix agrees with the sine
kernel to $7\times10^{-2}$. No theorem is claimed.
\end{nogobox}

\begin{nogobox}{The de Bruijn--Newman flow (heuristic remark)}
Paper~13~\cite[Thm.~2.1, Rem.~3.1]{P13}: $\dot L=-2\sum_{j\ne k}(y_j-y_k)^2/|z_j-z_k|^2$ for
$L=\sum(\Im z_j)^2$ under $\partial_tp=-\partial_{zz}p$, for every conjugate-symmetric
configuration of simple zeros; the identity uses no arithmetic of $\zeta$.
\end{nogobox}

\begin{nogobox}{Log-correlated picture (heuristic remark)}
Paper~05~\cite[Sections~6 and~7]{P05} (heuristic): the branching picture describes the maximum of
$\log|\zeta|$ on short intervals (upper bound and tightness proved by
Arguin--Bourgade--Radziwi{\l}{\l}~\cite{ABR20,ABR23}); it gives no information on zero location.
\end{nogobox}

\begin{nogobox}{Density versus absence}
A heuristic suggests that passing from zero density to the absence of off-line zeros
requires square-root cancellation. The theorem of de la Vall\'ee Poussin~\cite{dlVP96},
$\zeta(1+it)\ne0$, is an absence result on $\Re s=1$ obtained without such cancellation,
so this heuristic does not describe a general obstruction.
\end{nogobox}

\section{Heuristic Map of Stopping Points}
\label{sec:walls}
%% =====================================================================

\subsection{W1: PNT / Weil positivity}

\begin{wallbox}{W1, the classical equivalent forms}
The following are equivalent forms of RH (Proposition~\ref{prop:RH-equiv}):
$\psi(x) = x + O(x^{1/2+\eps})$ for all $\eps>0$; $B(f,f)\ge0$ for all
$f\in C_c^\infty(\mathbb R)$; vanishing of the negative index of the Weil form.
Approaches that reduce to one of them, such as direct attacks on the error term in the
prime number theorem or on Weil positivity, stop where these statements are open.
\end{wallbox}

\subsection{W2: realization in spaces of entire functions}

\begin{wallbox}{W2, de Branges realization}
A realization of the Weil form in a Pontryagin space of entire functions satisfying the
de Branges axioms is a space $H(E)$ with negative index $\ge\kV$
(Proposition~\ref{thm:rigidity}); the function $E_\xi$ is not a structure function, and
no realization with a structure function related to the zeros of $\xi$ is known.
Several programmes (Hilbert--P\'olya, de Branges canonical systems, Connes' trace
formula~\cite{Con99}, Burnol's de Branges spaces~\cite{Bur}) seek a Hilbert or Pontryagin
space of this kind attached to the zeros.
\end{wallbox}

\subsection{W3: statistics of the zeros}

\begin{wallbox}{W3, statistics}
Montgomery's pair-correlation theorem~\cite{Mon73} assumes RH, and the GUE conjectures
describe the statistics of the zeros; no implication from them to RH is known. The
numerical sine-kernel structure of Paper~12 and the identity of Paper~16 belong to this
statistical regime.
\end{wallbox}

\subsection{Chowla}

\begin{remark}
The Chowla conjecture concerns correlations of $\mu$ or $\lambda$; it is not known to
imply or be implied by RH, and it plays no role here.
\end{remark}

\section{Heuristic Synthesis}
\label{sec:synthesis}
%% =====================================================================

\subsection{Three kinds of data}

\begin{remark}[Heuristic synthesis]
\label{thm:synthesis}
The approaches surveyed use three kinds of data: (I) symmetry data (the invariance of the
zero set under $\rho\mapsto\bar\rho$ and $\rho\mapsto1-\bar\rho$), which holds for every
symmetric configuration of points, on or off the line, together with the arithmetic data of
the Euler product; (II) statistical data (gaps, pair correlation), which
describe distributions and do not locate individual zeros; (III) sign data, the vanishing
of a negative index, which is RH. Proposition~\ref{thm:sym-pos} shows that the symmetry alone does
not imply (III); whether the symmetry together with the Euler product implies (III) is the
open problem. We do not know a way to obtain (III) from (I) and (II); this is a description
of the approaches surveyed, not a theorem about all possible approaches.
\end{remark}

\begin{remark}
In this description, the approaches surveyed stop because they do not supply data of
type~(III).
\end{remark}

\subsection{Lower bounds and the negative index}

\begin{remark}[Heuristic]
A lower bound for an averaged quantity built from a test function $f$ is compatible with
off-line zeros whose ordinates lie where $f$ is small, as long as the zeros are regarded as
free parameters. This explains heuristically why such bounds, taken one test function at
a time, do not bound the negative index.
\end{remark}

\subsection{The function-field comparison}

\begin{remark}[Heuristic]
In the function-field case the local data (the Euler factors) and the global surface
$C\times C$ together give the Hodge index theorem and hence the Riemann hypothesis for
curves. For $\zeta$ the local data are available, and the global object is not.
\end{remark}

\subsection{Equivalent statements}

\begin{remark}
Each of the following is equivalent to RH: the criterion of von Koch~\cite{vK01},
$\psi(x)=x+O(x^{1/2+\eps})$ for every $\eps>0$; Weil's criterion~\cite{Weil52}; the vanishing of
the negative index of the Weil form. The approaches surveyed reduce to one of these
without proving it.
\end{remark}

\section{Open Problems}
\label{sec:open}
%% =====================================================================

\begin{enumerate}[label=\textbf{OP\arabic*.}]

\item \textbf{Unconditional window bounds.}
Prove $V_w(d,T)\ll dN(T)(\log T)^c$ without RH for some fixed $c$.
\emph{What is known}: $0\le V\le4dN^2$ unconditionally; $V_w\ll dN(T)$ under RH
(Paper~16).

\item \textbf{Class sums.}
Let $B(N)=\max_{t\in[N/10,N],\,k}|S_k(t;N)|^2$, the largest squared $\om$-class sum of
Paper~02~\cite[Def.~2.3]{P02}. Decide whether $B(N)\le C_\eps N^\eps$ for every $\eps>0$.
\emph{What is known}: this implies Lindel\"of (Paper~02~\cite[Cor.~4.2]{P02}); for the
class $k=1$ it is expected to be of the strength of RH; for the computed lower bounds of $B(N)$ at $N=10^4,\ 5\times10^4,\ 10^5,\ 1.5\times10^5$ the
ratio $\log B(N)/\log N$ lies between $0.41$ and $0.43$ (Paper~02, Section~8).

\item \textbf{A global object.}
Is there an arithmetic analogue of the surface $C\times C$ whose intersection theory gives
Weil positivity? No candidate is known.

\item \textbf{De Branges realizations.}
Does the Weil form admit a realization in a Pontryagin space of entire functions satisfying
\textup{(dB1)--(dB3)}, and if so, where are the zeros of its structure function?
\emph{What is known}: such a space is an $H(E)$ with negative index $\ge\kV$; $E_\xi$ is
not a structure function (Proposition~\ref{thm:rigidity}).

\item \textbf{Statistics of the zeros.}
Prove the pair-correlation asymptotics of Montgomery uniformly in $\alpha$ under RH
(the Goldston--Montgomery condition). No consequence for RH itself is claimed.

\item \textbf{Negative directions.}
Under the hypothesis of $m<\infty$ off-line quartets, is $\kV=m$?
\emph{What is known}: $\kV\le m$ (Proposition~\ref{thm:kappa}); for $J\ge2$, replacing a
double point $\half+i\gam$ by $\half\pm\delta+i\gam$ changes the zero-side Gram matrix by
$P(\gam,\delta)=\delta^2A(\gam)+R$, $\|R\|\le C(\gam,\delta)\delta^4$, with $A(\gam)$ of rank
$\le3$ and $\lambda_{\min}(A(\gam))<0$; for the quartet $\half\pm\delta\pm i\gam$ the change
is $P(\gam,\delta)+P(-\gam,\delta)$ (Proposition~\ref{thm:localization}).

\item \textbf{Eventual versus uniform positivity.}
Characterize the hierarchies of positivity statements for which eventual positivity
implies uniform positivity (Paper~15 treats elementary cases).

\end{enumerate}

\section{Conclusion}
\label{sec:conclusion}
%% =====================================================================

We have recalled what Papers~02--21 of the series prove and organised the points where the
corresponding approaches stop into a heuristic map with three ``walls''. The proved
statements are listed in Table~\ref{tab:new-results}; several are classical, and several
further statements of the series are numerical observations. The map and the
sign-versus-symmetry principle of Remark~\ref{thm:synthesis} are heuristic descriptions of
the approaches surveyed. Nothing here bears on the truth of RH.
