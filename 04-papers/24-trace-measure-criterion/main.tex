\title[Trace and measure reformulations]{Trace and measure reformulations of the Riemann Hypothesis\\
through the discrepancy measure of the on-line projection}
\author{David Alejandro Trejo Pizzo}
\thanks{Independent Researcher. \texttt{dtrejopizzo@gmail.com}}


\begin{document}
\begin{abstract}
Using the on-line projection $\zon$ of \cite{MomentRigidity}, for which
$\dnu=(|\zeta|^2-|\zon|^2)(\half+is)\,dm_\infty(s)\ge0$ and $\dnu=0\iff\RH$, we record several
reformulations of the Riemann Hypothesis as the vanishing of positive functionals of $\dnu$: a
trace against a positive kernel, a Nevanlinna transform, a Jensen excess, a relative entropy, an
$L^2$ distance and the total mass. All of them are trivial restatements of $\dnu=0$. We also give
a lower bound for the trace in terms of a single off-line zero, and we discuss the function-field
analogue informally. Nothing here gives information on $\RH$.
\end{abstract}

\section{Introduction}

Connes, Consani and Moscovici \cite{CCMprolate,CCMq,CCM} attach to $\zeta$ spectral data built on the
archimedean measure $dm_\infty(s)=(2\pi)^{-3/2}|\Gamma(\tfrac14+\tfrac{is}2)|^2\,ds$. Two
measures arise naturally from it. The \emph{full} measure $\dmfull=|\zeta(\half+is)|^2dm_\infty$
carries the actual modulus of $\zeta$ on the critical line; the \emph{on-critical} measure
$\dmon=|\zon(\half+is)|^2dm_\infty$ carries the modulus of the function $\zon$ obtained, as in
\cite{MomentRigidity}, by moving every zero of $\xi$ horizontally onto the critical line. If the
Riemann Hypothesis ($\RH$) holds, $\zeta=\zon$ and the two coincide; in general the difference
\[
\dnu=\dmfull-\dmon
\]
is a non-negative finite measure on $\mathbb R$ that vanishes exactly when $\RH$ holds. These
facts are proved in \cite[\S3]{MomentRigidity}; we recall them in
\S\ref{sec:measures}--\S\ref{sec:dominance} (Theorem~\ref{thm:dominance},
Corollary~\ref{cor:dnu}), stressing that the inequality $|\zon|\le|\zeta|$ depends on the
normalisation of the projection (Remark~\ref{rem:normalisation}).

In \cite{MomentRigidity} $\RH$ is shown equivalent to constancy in $n$ of
$\Delta_n=\int P_n^2\,\dnu$, where $P_n$ are the orthonormal (Meixner--Pollaczek) polynomials of
$dm_\infty$. There, $\RH\iff\Delta_n=0$ for a single $n$ is noted to be trivial, and the
determinacy of the $dm_\infty$ moment problem is due to Connes--Consani--Moscovici
\cite[Prop.~3.3, Thm.~5.2]{CCMprolate}, \cite[Prop.~3.1]{CCMq}.

The present paper lists further reformulations (Theorem~\ref{thm:eight}): the vanishing of a
Nevanlinna transform of $\dnu$, of a Jensen excess, of a relative entropy, of an $L^2$ distance,
and of the trace $\Tlam=\int\Wlam\,\dnu$ against a positive kernel $\Wlam$. Each condition is a
positive functional of the non-negative measure $\dnu$ (or a sum of positive terms over off-line
zeros) and vanishes iff $\dnu=0$; all equivalences are therefore trivial consequences of
Corollary~\ref{cor:dnu}. In particular $\Tlam=0$ for a \emph{single} $\lambda$ is already
equivalent to $\RH$ (Theorem~\ref{thm:trace-crit}). We give an explicit lower bound for $\Tlam$
in terms of one off-line zero (Proposition~\ref{prop:lower}). Finally, \S\ref{sec:weil} defines
the analogue of $\dnu$ for curves over finite fields, where it vanishes by Weil's theorem
\cite{Weil48}, and comments heuristically on the comparison. Nothing in the paper assumes or
proves $\RH$.

\section{The two measures and the density \texorpdfstring{$R$}{R}}\label{sec:measures}

Let $\xi(s)=\half s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)$ and $G(s)=\half s(s-1)\pi^{-s/2}\Gamma(s/2)$.
The zeros of $\xi$ off the critical line, if any, occur in quadruples
$Q_j=\{\half\pm\delta_j\pm i\gamma_j\}$ with $\delta_j\in(0,\half)$, $\gamma_j>0$, $j\in J$, and
$J=\emptyset$ iff $\RH$.

\begin{definition}[Full and on-critical measures]\label{def:measures}
Let $dm_\infty(s)=(2\pi)^{-3/2}|\Gamma(\tfrac14+\tfrac{is}2)|^2\,ds$
\cite[Def.~2.1]{MomentRigidity}. For $j\in J$ put
\[
R_j(s)=\frac{\bigl((s-\half)^2+\gamma_j^2\bigr)^2}
{\bigl((s-\half)^2-(\delta_j+i\gamma_j)^2\bigr)\bigl((s-\half)^2-(\delta_j-i\gamma_j)^2\bigr)},
\qquad \xon:=\xi\prod_{j\in J}R_j,
\]
and $|\zon(\half+is)|:=|\xon(\half+is)|/|G(\half+is)|$ \cite[Def.~3.1,
Lemma~3.2]{MomentRigidity}; the product converges locally uniformly off $\bigcup_jQ_j$ and $\xon$
is entire, with zeros the critical zeros of $\xi$ together with $\half\pm i\gamma_j$ (double).
The full and on-critical measures are
\[
\dmfull=|\zeta(\half+is)|^2\,dm_\infty,\qquad
\dmon=|\zon(\half+is)|^2\,dm_\infty .
\]
Off the discrete set of zeros of $\zon(\half+is)$ put $R(s)=|\zeta(\half+is)/\zon(\half+is)|^2$;
then
\[
\dnu=\dmfull-\dmon=(R-1)\,\dmon .
\]
\end{definition}

\section{The dominance theorem}\label{sec:dominance}

\begin{theorem}[Dominance of $\zeta$ over $\zon$ \cite{MomentRigidity}]\label{thm:dominance}
For every real $t$ that is not an ordinate of a zero of $\zon(\half+it)$,
\[
R(t)=\prod_{j\in J}\frac{(\delta_j^2+(t-\gamma_j)^2)^2(\delta_j^2+(t+\gamma_j)^2)^2}
{(t-\gamma_j)^4(t+\gamma_j)^4}\ \ge 1,
\]
each factor being $>1$. Hence $R\equiv1$ if and only if $\RH$ holds.
\end{theorem}
\begin{proof}
This is \cite[Prop.~3.4]{MomentRigidity}; we recall the computation. At $s=\half+it$ one has
$(s-\half)^2=-t^2$, the two denominator factors of $R_j$ are complex conjugate, and the identity
$(\gamma^2-\delta^2-t^2)^2+4\delta^2\gamma^2=((t-\gamma)^2+\delta^2)((t+\gamma)^2+\delta^2)$ gives
\[
R_j(\half+it)=\frac{(t-\gamma_j)^2(t+\gamma_j)^2}{(\delta_j^2+(t-\gamma_j)^2)(\delta_j^2+(t+\gamma_j)^2)}
\in[0,1).
\]
Since $|G|$ cancels, $R(t)=|\xi/\xon|^2(\half+it)=\prod_j|R_j(\half+it)|^{-2}$, which is the
displayed product. Each factor exceeds $1$ because $\delta_j>0$. Convergence of the product is
\cite[Lemma~3.2]{MomentRigidity}.
\end{proof}

\begin{corollary}[$\dnu$ is a non-negative invariant for $\RH$]\label{cor:dnu}
$\dnu=(R-1)\,\dmon\ge0$, and $\dnu=0\iff\RH$. Moreover $\dnu$ is even, finite, with all moments
finite, and has a continuous density $\tilde g(s)\,w(s)$ with respect to Lebesgue measure, where
$\tilde g=|\zeta(\half+is)|^2-|\zon(\half+is)|^2$ and $w$ is the density of $dm_\infty$.
\end{corollary}
\begin{proof}
Non-negativity and the equivalence are Theorem~\ref{thm:dominance} (if $J\ne\emptyset$, then
$\tilde g>0$ at every $s$ which is not a zero of $\zon(\half+is)$, since $|\zon|^2>0$ there and
$R>1$). Evenness, continuity and finiteness of moments are \cite[Lemma~3.2,
Prop.~3.6]{MomentRigidity}.
\end{proof}

\begin{remark}[The curve case]\label{rem:curve}
For a smooth projective curve over a finite field the analogous measure vanishes identically,
because every zero of its zeta function lies on the critical circle (Weil \cite{Weil48}); see
Theorem~\ref{thm:ff}. For $\zeta$ we know only $\dnu\ge0$.
\end{remark}

\begin{remark}[Dependence on the normalisation]\label{rem:normalisation}
The inequality $R\ge1$ uses the normalisation of $R_j$ above, for which $R_j(s)\to1$ as
$|s|\to\infty$. If instead $\zon$ were defined by replacing the zeros in the canonical Hadamard
product with factors $(1-s/\rho)e^{s/\rho}$, the factor of one quadruple would be multiplied by
a positive constant (the moduli of the normalisations $1/\rho$ and of the exponentials are
constant on the line $\Re s=\half$), and the resulting factor tends, as $|t|\to\infty$, to
\[
\frac{b_j^4}{(a_j^2-\delta_j^2)^2}\,
\exp\Bigl(-\frac{2\delta_j^2(3b_j+\delta_j^2-1)}{b_j(a_j^2-\delta_j^2)}\Bigr),\qquad
b_j=\tfrac14+\gamma_j^2,\quad a_j=b_j+\delta_j^2;
\]
both the rational constant and the exponential are $<1$ when $\gamma_j^2>\tfrac14$. For instance, with $\delta=0.3$, $\gamma=20$ this factor is
about $0.99914$ at $t=100$ and $0.99910$ at $t=10^4$, so the factor is $<1$ for large $|t|$: the factorwise inequality of Theorem~\ref{thm:dominance} fails, and so does $R\ge1$ itself when there are finitely many off-line quadruples.
\end{remark}

\section{Trivially equivalent reformulations}\label{sec:eight}

Fix, for $\lambda>0$, the kernel $\Wlam(s)=\lambda^2/(\lambda^2+s^2)$ (a Poisson-type kernel;
$0<\Wlam\le1$, continuous, and $\Wlam\uparrow1$ as $\lambda\to\infty$), and put
\[
\Tlam=\int_{\mathbb R}\Wlam\,\dnu .
\]
Everything below except the identity $\sup_\lambda\Tlam=\nu(\mathbb R)$ in the proof of (8)
holds for any continuous kernel with $0<\Wlam\le C$ (item (8) remains equivalent, since
$\sup_\lambda\Tlam=0$ iff every $\Tlam=0$).

\begin{theorem}[Equivalent reformulations]\label{thm:eight}
The following are equivalent.
\begin{enumerate}[\upshape(1)]
\item \textup{(RH)} every nontrivial zero of $\zeta$ has $\Re s=\half$;
\item \textup{(Zero measure)} $\dnu=0$;
\item \textup{(Nevanlinna)} $\mathcal G(z)=\int_{\mathbb R}\frac{1+sz}{s-z}\,\dnu(s)$ vanishes
identically on $\{\Im z>0\}$;
\item \textup{(Trace)} $\Tlam=0$ for some \textup{(}equivalently, every\textup{)} $\lambda>0$;
\item \textup{(Jensen excess)} $E_J:=\sum_{\Re\rho>1/2}\log(|\rho|/|\rho-1|)=0$, the sum over
zeros of $\xi$ with multiplicity;
\item \textup{($L^2$ distance)} $\widetilde\delta^2:=\int_{\mathbb R}\bigl(|\zeta(\half+is)|-
|\zon(\half+is)|\bigr)^2dm_\infty(s)=0$;
\item \textup{(Relative entropy)} $\mathrm{KL}(\dmfull\,\|\,\dmon)=0$;
\item \textup{(Total mass)} $\sup_{\lambda>0}\Tlam=0$.
\end{enumerate}
All equivalences are trivial restatements of Corollary~\ref{cor:dnu}: each condition is a
positive functional of the non-negative measure $\dnu$ \textup{(}or, for \textup{(5)}, a sum of
positive terms over off-line zeros\textup{)} and vanishes iff $\dnu=0$.
\end{theorem}
\begin{proof}
$(1)\Leftrightarrow(2)$ is Corollary~\ref{cor:dnu}.

$(2)\Leftrightarrow(3)$: $\dnu$ is finite with finite second moment, so $\mathcal G$ is defined.
Since $\Im\frac{1+sz}{s-z}=\frac{(1+s^2)\Im z}{|s-z|^2}$, one has $\Im\mathcal G(i)=\nu(\mathbb R)$;
so $\mathcal G\equiv0$ forces $\nu(\mathbb R)=0$, i.e.\ $\dnu=0$ (alternatively, by Stieltjes
inversion $\Im\mathcal G$ determines $(1+s^2)\dnu$). The converse is clear.

$(2)\Leftrightarrow(4)$: if $\dnu=0$ every $\Tlam$ vanishes. Conversely, $\Wlam\tilde g w$ is
continuous and $\ge0$, so $\Tlam=0$ for one $\lambda$ forces $\Wlam\tilde g w\equiv0$, hence
$\tilde g\equiv0$ since $\Wlam w>0$.

$(1)\Leftrightarrow(5)$: for $\Re\rho>\half$ one has $|\rho|>|\rho-1|$, so each term is
positive; the series converges since $\log(|\rho|/|\rho-1|)=O(\delta/|\rho|^2)$ and
$\sum_\rho|\rho|^{-2}<\infty$. By the symmetry $\rho\mapsto1-\bar\rho$, zeros with
$\Re\rho>\half$ exist iff $\RH$ fails.

$(2)\Leftrightarrow(6)$: $|\zeta|\ge|\zon|$ on the line, so $\widetilde\delta^2=0$ iff
$|\zeta|=|\zon|$ $dm_\infty$-a.e., iff (by continuity) $\tilde g\equiv0$.

$(2)\Leftrightarrow(7)$: $\dmfull=R\,\dmon$, so $\mathrm{KL}(\dmfull\|\dmon)=\int R\log R\,\dmon
\in[0,\infty]$, the integrand being $\ge0$ as $R\ge1$; it vanishes iff $R=1$ $\dmon$-a.e., iff
$\dnu=(R-1)\dmon=0$.

$(2)\Leftrightarrow(8)$: by monotone convergence $\sup_\lambda\Tlam=\lim_{\lambda\to\infty}\Tlam
=\nu(\mathbb R)$, which vanishes iff $\dnu=0$.
\end{proof}

\section{The trace and its lower bound}\label{sec:trace}

\begin{proposition}[Lower bound under $\neg\RH$]\label{prop:lower}
Suppose $\neg\RH$. Then $\Tlam>0$ for every $\lambda>0$, and for each off-line quadruple $Q_j$,
\[
\Tlam\ \ge\ \int_{\mathbb R}\Wlam(s)\,|\zon(\half+is)|^2\,
\frac{2\delta_j^2}{(s-\gamma_j)^2}\,dm_\infty(s)\ >0 .
\]
\end{proposition}
\begin{proof}
Off the zeros of $\zon(\half+is)$, $\tilde g=|\zon|^2(R-1)$. With $a=\delta_j^2/(s-\gamma_j)^2$,
all factors of the product in Theorem~\ref{thm:dominance} are $\ge1$ and the $j$-th is
$\ge(1+a)^2$, so $R-1\ge(1+a)^2-1=2a+a^2\ge2a$. Integrating against $\Wlam\,dm_\infty$ gives the
inequality (the integrand on the right is integrable: near $\gamma_j$ the double zero of $\zon$
cancels the pole, and globally it is bounded by $C\,\tilde g\,w$). It is positive because the
integrand is continuous, $\ge0$, and positive off a discrete set.
\end{proof}

Since $\dnu\ge0$ and $\Wlam>0$, no cancellation occurs in $\Tlam$, and a single value
$\Tlam=0$ already forces $\dnu=0$.

\begin{remark}[No completeness hypothesis is needed]\label{def:completeness}
One might try to recover $\dnu$ from the family $\{\Tlam\}$ through a totality property of the
kernels $\Wlam$ in $L^2(\dnu)$. This is unnecessary: because $\dnu\ge0$ has a continuous density
and $\Wlam>0$, the vanishing of one $\Tlam$ already gives $\dnu=0$.
\end{remark}

\begin{theorem}[Trace criterion]\label{thm:trace-crit}
For any $\lambda>0$, $\Tlam=0\iff\RH$. This is a trivial consequence of
Corollary~\ref{cor:dnu} and $\Wlam>0$.
\end{theorem}
\begin{proof}
This is $(1)\Leftrightarrow(4)$ of Theorem~\ref{thm:eight}.
\end{proof}

\begin{remark}[What is used]\label{rem:notused}
Theorem~\ref{thm:trace-crit} uses only $\dnu\ge0$ and the scalar positivity $\Wlam>0$; no
positivity of any kernel as a quadratic form is involved. Like $\Delta_n=0$ for a single $n$ in
\cite{MomentRigidity}, it is a restatement of $\dnu=0$; the non-trivial content of
\cite{MomentRigidity} is that constancy of $\Delta_n$ in $n$, with unknown value, already forces
$\dnu=0$.
\end{remark}

\section{The function-field mirror (heuristic)}\label{sec:weil}

Let $C$ be a smooth projective curve of genus $g$ over $\mathbb F_q$, with zeta function
$Z_C(u)=P(u)/((1-u)(1-qu))$, $P(u)=\prod_{i=1}^{2g}(1-\alpha_iu)$. Define the on-circle
projection $P_{\mathrm{on}}(u)=\prod_i(1-\alpha_i'u)$ with $\alpha_i'=q^{1/2}\alpha_i/|\alpha_i|$
(each zero $\alpha_i^{-1}$ moved radially to $|u|=q^{-1/2}$), and
\[
\dnu^C(t)=\bigl(|P(q^{-1/2-it})|^2-|P_{\mathrm{on}}(q^{-1/2-it})|^2\bigr)\,dt,\qquad
t\in[0,2\pi/\log q).
\]

\begin{theorem}[Function-field analogue vanishes]\label{thm:ff}
$\dnu^C=0$.
\end{theorem}
\begin{proof}
By Weil's theorem \cite{Weil48}, $|\alpha_i|=q^{1/2}$ for all $i$, so $\alpha_i'=\alpha_i$,
$P_{\mathrm{on}}=P$ and $\dnu^C=0$.
\end{proof}

The following comparison is heuristic. In modern language, the proof of Weil's theorem rests on three structures, of which $\zeta$
has analogues of the first two but not the third:
\begin{enumerate}[(P1)]
\item a finite-dimensional cohomology $H^1(C)$ carrying the zeros as Frobenius eigenvalues
(\emph{for $\zeta$}: candidate spectral realizations exist \cite{bgConnes99,CCM}, but they are
infinite-dimensional);
\item a Lefschetz trace formula relating the eigenvalues to point counts (\emph{for $\zeta$}: the
explicit formula plays this role);
\item a positivity statement (the Castelnuovo--Severi inequality, i.e.\ the Hodge index theorem
on $C\times C$; equivalently positivity of the Rosati involution) that forces
the Frobenius eigenvalues onto the critical circle (\emph{for $\zeta$}: no analogue is known).
\end{enumerate}

\begin{remark}
In the curve case the positivity (P3) gives $\dnu^C=0$; for $\zeta$ the projection gives only
$\dnu\ge0$. Any of the reformulations of \S\ref{sec:eight}, and the moment rigidity of
\cite{MomentRigidity}, asks for the missing vanishing.
\end{remark}

\section{The Hadamard barrier}\label{sec:barrier}

\begin{definition}[Hadamard barrier]\label{def:barrier}
By the \emph{Hadamard barrier} we mean the gap between two facts: (a) the comparison of $\xi$
with its on-line projection yields, unconditionally, $R\ge1$, i.e.\ $\dnu\ge0$; (b) $\dnu=0$ is
equivalent to $\RH$.
\end{definition}

\begin{remark}[The reformulations are restatements]\label{prop:barrier}
$\dnu=0\iff\RH$ is a restatement of $\RH$, and every criterion of \S\ref{sec:eight} is
equivalent to it trivially. A non-negative even finite measure need not vanish, so $\dnu\ge0$
alone gives no information on whether $\dnu=0$.
\end{remark}

\section{Conclusion}

Using the on-line projection of \cite{MomentRigidity}, the measure
$\dnu=(|\zeta|^2-|\zon|^2)dm_\infty$ is non-negative and vanishes iff $\RH$
(Theorem~\ref{thm:dominance}, Corollary~\ref{cor:dnu}, both from \cite{MomentRigidity}). The
reformulations of Theorem~\ref{thm:eight}, including the single-$\lambda$ trace criterion
(Theorem~\ref{thm:trace-crit}), are trivial restatements of $\dnu=0$; Proposition~\ref{prop:lower}
gives an explicit lower bound for $\Tlam$ in terms of one off-line zero. The function-field
analogue vanishes by Weil's theorem (Theorem~\ref{thm:ff}). The paper neither assumes nor proves
the Riemann Hypothesis.
