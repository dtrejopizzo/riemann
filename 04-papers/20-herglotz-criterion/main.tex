\title[A Herglotz positivity criterion]{A Herglotz positivity criterion for the Riemann hypothesis}
\author{David Alejandro Trejo Pizzo}
\thanks{Independent Researcher. \texttt{dtrejopizzo@gmail.com}}


\begin{document}
\begin{abstract}
We give an expository account of the known criterion (Hinkkanen, Lagarias) that
the Riemann hypothesis holds if and only if $\Re\,\xi'/\xi(s)>0$ for $\Re s>\half$,
equivalently that $\sigma\mapsto|\xi(\sigma+i\tau)|$ is increasing for
$\sigma>\half$ (Sondow--Dumitrescu). We record the elementary identity
$\Re\,\xi'/\xi(\half+i\tau)=0$ at every $\tau$ with $\xi(\half+i\tau)\neq0$, its
consequence for $\log|\xi|$ near the critical line, the Nevanlinna-function form of
the criterion, and a few numerical evaluations. Nothing here goes beyond the known
equivalence.
\end{abstract}

\section{Introduction}
Let
\[
\xi(s)=\half s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)
\]
be the completed Riemann zeta function: entire, of order $1$, real on $\mathbb R$,
with $\xi(s)=\xi(1-s)$, whose zeros are exactly the nontrivial zeros
$\rho=\beta+i\gamma$ of $\zeta$, all in the strip $0<\beta<1$. The Riemann
Hypothesis is $\beta=\half$ for all $\rho$.

The Weil quadratic form expresses $\RH$ as a positivity statement; a Kre\u\i n-space
formulation of its zero side is given in~\cite{RP8}. This note records a simpler,
pointwise positivity equivalent to $\RH$.

\textbf{The criterion.} $\RH$ holds if and only if
\begin{equation}\label{eq:crit}
\Re\,\frac{\xi'}{\xi}(\sigma+i\tau)>0\qquad\text{for all }\sigma>\half,\ \text{all }\tau\in\mathbb R.
\end{equation}
This equivalence is due to Hinkkanen~\cite{Hinkkanen} and Lagarias~\cite{Lagarias99};
in the equivalent form ``$|\xi(\sigma+i\tau)|$ is strictly increasing in $\sigma$ for
$\sigma>\half$'' it appears in Sondow--Dumitrescu~\cite{SondowDumitrescu}. It is
closely related to the Laguerre--P\'olya and Hermite--Biehler descriptions of
$\Xi(z):=\xi(\half+iz)$ and to de Branges' theory~\cite{deBranges,Lagarias}.
The paper is expository: we give short proofs of the criterion, of the elementary
boundary identity on the critical line, and of the Nevanlinna-function form, and we
include a few numerical evaluations. It is a \emph{reformulation}, equivalent to
$\RH$, not a proof.

\section{The criterion}
\begin{proposition}[The criterion is $\RH$]\label{prop:crit}
Writing $\Re\,\xi'/\xi(\sigma+i\tau)=\sum_\rho\frac{\sigma-\beta_\rho}{|\sigma+i\tau-\rho|^2}$
(symmetric Hadamard sum over the zeros), the positivity~\eqref{eq:crit} holds if
and only if there is no zero with $\beta>\half$ — i.e.\ if and only if $\RH$.
\end{proposition}
\begin{proof}
The sum converges absolutely, since $|\sigma-\beta_\rho|\le\sigma+1$ and
$\sum_\rho|\rho|^{-2}<\infty$. If $\RH$ holds, every term has
$\sigma-\beta_\rho=\sigma-\half>0$ for $\sigma>\half$, so $\Re\,\xi'/\xi>0$.
Conversely, let $\rho_0=\beta_0+i\gamma_0$ with $\beta_0>\half$. Then $\xi'/\xi$ has
a simple pole at $\rho_0$ with residue $m\ge1$ (the multiplicity), so
$\xi'/\xi(s)=m/(s-\rho_0)+O(1)$ near $\rho_0$. At $s=\rho_0-\delta$ with
$0<\delta<\beta_0-\half$ this gives $\Re\,\xi'/\xi(s)=-m/\delta+O(1)\to-\infty$ as
$\delta\downarrow0$, contradicting~\eqref{eq:crit}.
\end{proof}

\begin{remark}
A function with positive real part on $\{\Re s>\half\}$ is holomorphic there. Since
the zeros of $\xi$ are poles of $\xi'/\xi$, \eqref{eq:crit} excludes zeros with
$\beta>\half$, and hence, by the symmetry $\rho\mapsto1-\bar\rho$, zeros with
$\beta<\half$.
\end{remark}

\section{The unconditional boundary identity}
The following elementary fact does not depend on the location of the zeros.
\begin{theorem}[Boundary identity]\label{thm:boundary}
For all real $\tau$ such that $\xi(\half+i\tau)\neq0$,
\[
\Re\,\frac{\xi'}{\xi}\!\left(\half+i\tau\right)=0 .
\]
\end{theorem}
\begin{proof}
Differentiating $\xi(s)=\xi(1-s)$ gives $\xi'/\xi(s)=-\,\xi'/\xi(1-s)$. On
$s=\half+i\tau$ we have $1-s=\half-i\tau=\bar s$, and since $\xi$ has real Taylor
coefficients, $\xi'/\xi(\bar s)=\overline{\xi'/\xi(s)}$. Therefore
$\xi'/\xi(s)=-\overline{\xi'/\xi(s)}$, whence $\Re\,\xi'/\xi(s)=0$. (See also Appendix~\ref{app:code}.)
\end{proof}

\begin{corollary}[The critical line is a critical line of $\log|\xi|$ in $\sigma$]\label{cor:min}
For all real $\tau$ such that $\xi(\half+i\tau)\neq0$,
$\partial_\sigma\log|\xi(\sigma+i\tau)|=\Re\,\xi'/\xi(\sigma+i\tau)$ vanishes at
$\sigma=\half$, and
\[
\partial_\sigma^2\log|\xi(\sigma+i\tau)|\Big|_{\sigma=\half}=\Re\,(\xi'/\xi)'(\half+i\tau)
=\sum_\rho\frac{(\tau-\gamma_\rho)^2-(\half-\beta_\rho)^2}{|\half+i\tau-\rho|^4},
\]
the sum over zeros counted with multiplicity. Under $\RH$ this equals
$\sum_\rho(\tau-\gamma_\rho)^{-2}>0$, so for such $\tau$ the function
$\sigma\mapsto|\xi(\sigma+i\tau)|$ has a strict local minimum at $\sigma=\half$.
\end{corollary}

\section{Harmonic-function structure}
$\Re\,\xi'/\xi$ is harmonic in $\{\sigma>\half\}$ away from the zeros, it vanishes
on the line $\sigma=\half$ away from the ordinates of zeros on that line
(Theorem~\ref{thm:boundary}), and $\Re\,\xi'/\xi(\sigma+i\tau)\to+\infty$ as
$\sigma\to+\infty$ for fixed $\tau$ (the $\Gamma$-factor dominates).

\begin{proposition}[Interior poles and the sign]\label{prop:minprinciple}
If $\xi$ has no zero in $\{\sigma>\half\}$, then $\Re\,\xi'/\xi>0$ throughout
$\{\sigma>\half\}$. If $\rho_0$ is a zero with $\beta_0>\half$, then
$\Re\,\xi'/\xi$ takes arbitrarily large negative values near $\rho_0$. Hence
\eqref{eq:crit} holds if and only if $\xi$ has no zero in $\{\sigma>\half\}$.
\end{proposition}
\begin{proof}
If there is no zero with $\beta>\half$, then by the functional equation there is
none with $\beta<\half$, so every term $(\sigma-\half)/|s-\rho|^2$ of the Hadamard
sum of Proposition~\ref{prop:crit} is positive for $\sigma>\half$. The second
statement is the converse part of the proof of Proposition~\ref{prop:crit}.
\end{proof}

\begin{remark}
One might hope to derive positivity from the boundary values and the growth in
$\sigma$ by a minimum principle on the half-plane. This is only heuristic: it
would require control of $\Re\,\xi'/\xi$ as $|\tau|\to\infty$ in
$\half<\sigma<1$ and near the boundary singularities, and the harmonic function
$-(\sigma-\half)$, which vanishes on the boundary, shows that boundary vanishing
alone is not enough.
\end{remark}

\begin{remark}
Theorem~\ref{thm:boundary} gives the boundary values of $\Re\,\xi'/\xi$
unconditionally. Proving~\eqref{eq:crit} in $\half<\sigma<1$ without assuming
the absence of zeros there is equivalent to $\RH$.
\end{remark}

\section{The Nevanlinna-function form}
With $s=\half-iz$, put $m(z)=i\,\xi'/\xi(\half-iz)$. For $\Im z>0$ we have
$\Re s=\half+\Im z>\half$ and $\Im m(z)=\Re\,\xi'/\xi(s)$, so~\eqref{eq:crit} says
exactly that $m$ is a Nevanlinna function (holomorphic with positive imaginary part)
on $\Im z>0$. Under $\RH$, the zeros $\half+i\gamma$ give real points $z=-\gamma$,
the set of ordinates is symmetric, and
\[
m(z)=a+bz+\sum_\gamma m_\gamma\Big(\frac1{\gamma-z}-\frac{\gamma}{1+\gamma^2}\Big),
\qquad a\in\mathbb R,\ b\ge0,
\]
with $m_\gamma$ the multiplicities; $m$ is then the Weyl function of multiplication
by $t$ on $L^2(\sum_\gamma m_\gamma\delta_\gamma)$. Every Nevanlinna function has
such a realization built from its own measure~\cite{deBranges}; here the
realization is built from the zeros themselves and gives no independent route to
$\RH$.

\section{Remarks}
De Branges' sufficient positivity condition fails for $\zeta$
(Conrey--Li~\cite{ConreyLi}).

\section*{Conclusion}
$\RH$ is equivalent to the positivity of $\Re\,\xi'/\xi$ on the right half-plane
(Hinkkanen, Lagarias). Unconditionally, $\Re\,\xi'/\xi$ vanishes on the critical
line away from ordinates of zeros, so $|\xi|$ is stationary in $\sigma$ there. The
remaining content is the absence of zeros in $\half<\sigma<1$.

\section{Reproducible computations}\label{app:code}
\subsection*{A.1 Sample values of the real part of the logarithmic derivative (Prop.~\ref{prop:crit})}
\begin{lstlisting}
import mpmath as mp
mp.mp.dps = 25
def xi(s): return mp.mpf('0.5')*s*(s-1)*mp.pi**(-s/2)*mp.gamma(s/2)*mp.zeta(s)
def xilog(s): return mp.diff(lambda z: mp.log(xi(z)), s)   # xi'/xi
for sigma in ['1.0','0.7','0.55','0.501']:
    sg=mp.mpf(sigma)
    vals=[mp.re(xilog(sg+1j*mp.mpf(t))) for t in [0,10,14.13,20,30,50]]
    print(sigma, "min Re xi'/xi =", mp.nstr(min(vals),5))   # all > 0 at these points
\end{lstlisting}
