\title[The second $\sigma$-derivative of $\log|\xi|$]{The second $\sigma$-derivative of $\log|\xi|$\\
  on the critical line}
\author{David Alejandro Trejo Pizzo}
\thanks{Independent Researcher. \texttt{dtrejopizzo@gmail.com}}


\begin{document}

\begin{abstract}
We record the Hadamard-product formula for $\partial^2_\sigma\log|\xi(\sigma+i\gamma)|$,
show that it is positive at $\sigma=1/2$ off the ordinates of the zeros under the
Riemann Hypothesis (RH), compute its gamma-factor and polynomial parts explicitly, and
discuss why the converse remains open. Writing
$\mathcal C(\gamma)=\partial^2_\sigma\log|\xi(\sigma+i\gamma)|\big|_{\sigma=1/2}$, an off-line
zero $\rho_0=\tfrac12+b_0+i\gamma_0$ contributes, together with $1-\bar\rho_0$, the amount
$-2/b_0^2$ to $\mathcal C(\gamma_0)$; whether $\mathcal C>0$ on $\mathbb R$ implies RH is
open, and the analogous implication fails for general real entire functions of order one.
We also relate this positivity to the de~Bruijn--Newman constant and to the negative index
of the Weil quadratic form, and we write $\mathcal C$ as the sum of an explicit analytic
part, which is negative near $t=0$ and behaves like $7/(4t^2)$ for large $t$, and an
arithmetic part coming from $\zeta(\tfrac12+it)$.
\end{abstract}

\section{Introduction}
\label{sec:intro}
%------------------------------------------------------------------

The Riemann Hypothesis asserts that every non-trivial zero $\rho$ of the Riemann
$\zeta$-function satisfies $\Re(\rho) = 1/2$. This paper studies the function
\[
  \mathcal{C}(\gamma) := \frac{\partial^2}{\partial\sigma^2}\log|\xi(\sigma+i\gamma)|\bigg|_{\sigma=1/2},
  \qquad \gamma\in\mathbb R,\ \xi(\tfrac12+i\gamma)\neq0,
\]
the second $\sigma$-derivative of $\log|\xi|$ on the critical line. By harmonicity of
$\log|\xi|$ away from its zeros, $\mathcal C=-(\log|f|)''$ with $f(t)=\xi(1/2+it)$.

\subsection*{Contents}

\begin{enumerate}
  \item \textbf{Hadamard formula (Theorem~\ref{thm:curvature}).} The classical formula
    $\partial^2_{\sigma}\log|\xi(s)|=\sum_\rho\Re\bigl(-(s-\rho)^{-2}\bigr)$, written in
    Cartesian coordinates.
  \item \textbf{One direction (Theorem~\ref{thm:rh-curvature}).} Under RH,
    $\mathcal C(\gamma)=\sum_\rho(\gamma-\gamma_\rho)^{-2}>0$ off the ordinates. Conversely, an
    off-line zero $\rho_0=\tfrac12+b_0+i\gamma_0$ contributes $-2/b_0^2$ to $\mathcal C(\gamma_0)$
    (with $1-\bar\rho_0$), and $\mathcal C(\gamma_0)<0$ whenever the other zeros contribute
    less than $2/b_0^2$. Whether $\mathcal C>0$ on $\mathbb R$ implies RH is open; the
    analogous statement for general real entire functions is false
    (Remark~\ref{rem:counterexample}).
  \item \textbf{Relation to known criteria (Theorem~\ref{thm:lp}).} The conditions
    $\Lambda=0$, $\kappa(Q)=0$, $f\in\LP$ and RH are equivalent (by known results), and imply
    $\mathcal C>0$.
  \item \textbf{Sector decomposition (Theorem~\ref{thm:sectors}).} Closed-form expressions for
    the contributions of the polynomial and gamma factors of $\xi$ to $\mathcal C$, with
    their asymptotics (Proposition~\ref{prop:gana}).
\end{enumerate}

\subsection*{Relation to known results}

The formula of Theorem~\ref{thm:curvature} is immediate from the Hadamard product, and the
positivity of $\mathcal C$ under RH is a one-line consequence. Positivity criteria of this
type are known: Lagarias~\cite{Lagarias1999} proved that RH holds if and only if
$\Re(\xi'/\xi)(s)>0$ for $\Re s>1/2$. The equivalence $\mathrm{RH}\iff f\in\LP$, $f(t)=\xi(1/2+it)$,
goes back to P\'olya and is the starting point of de~Bruijn~\cite{deBruijn1950} and
Newman~\cite{Newman1976}; $\Lambda \geq 0$ is due to Rodgers--Tao~\cite{RodgersTao2020}.
The equivalence $\mathrm{RH}\iff\kappa(Q)=0$ for the Weil form is Weil's positivity theorem,
in the form recorded in \cite[Thm.~2.2, Def.~2.3]{P25}. What this paper adds is the explicit
computation of the analytic part of $\mathcal C$ (Theorem~\ref{thm:sectors},
Proposition~\ref{prop:gana}) and the discussion of the converse.

\subsection*{Notation}
Throughout, $\xi(s) = \frac{1}{2}s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)$ is the completed
zeta function; non-trivial zeros of $\zeta$ are written $\rho = \beta_\rho + i\gamma_\rho$
with $0 < \beta_\rho < 1$, counted with multiplicity; $\psi^{(1)}$ denotes the trigamma
function. We write $f(t) := \xi(1/2+it)$ for $t \in \mathbb{R}$.

%------------------------------------------------------------------
\section{The Hadamard formula}
\label{sec:curvature}
%------------------------------------------------------------------

\subsection{Setup}

The function $\xi$ is entire of order one, and $\sum_\rho|\rho|^{-2}<\infty$. Its Hadamard
product is
\begin{equation}\label{eq:hadamard}
  \xi(s) = e^{A+Bs}\prod_\rho\!\left(1-\frac{s}{\rho}\right)e^{s/\rho}
\end{equation}
for constants $A,B$. Write $\mathcal{L}(s) = \log|\xi(s)|$, which is harmonic away from the
zeros. Taking real parts of the logarithm,
\begin{equation}\label{eq:log-xi}
  \mathcal{L}(s) = \Re(A+Bs) + \sum_\rho \Re\!\left[\log\!\left(1-\frac{s}{\rho}\right)+\frac{s}{\rho}\right].
\end{equation}

\begin{theorem}[Hadamard formula]\label{thm:curvature}
For $s = \sigma + i\gamma$ not a zero of $\xi$,
\begin{equation}\label{eq:curvature-cartesian}
  \frac{\partial^2}{\partial\sigma^2}\,\mathcal{L}(\sigma+i\gamma)
  \;=\;\sum_\rho\Re\frac{-1}{(s-\rho)^2}\;=\;
  \sum_\rho \frac{(\gamma-\gamma_\rho)^2 - (\sigma-\beta_\rho)^2}{|s-\rho|^4}.
\end{equation}
\end{theorem}

\begin{proof}
The linear terms $\Re(A+Bs)$ and $\Re(s/\rho)$ have vanishing second $\sigma$-derivative.
For a single factor, $\partial_\sigma$ acts on holomorphic functions as $d/ds$, so
\[
  \frac{\partial^2}{\partial\sigma^2}\Re\log\!\left(1-\frac{s}{\rho}\right)
  = \Re\frac{-1}{(s-\rho)^2}
  = \frac{(\gamma-\gamma_\rho)^2-(\sigma-\beta_\rho)^2}{|s-\rho|^4}.
\]
On a compact set $K$ avoiding the zeros, $|s-\rho|\ge c_K|\rho|$ for all but finitely many
$\rho$, so $\sum_\rho|s-\rho|^{-2}$ converges uniformly on $K$ because
$\sum_\rho|\rho|^{-2}<\infty$. Hence the series of second derivatives converges uniformly on
$K$; since the series of logarithms in~\eqref{eq:log-xi} converges locally uniformly as well,
term-by-term differentiation is justified.
\end{proof}

\begin{remark}
In polar form, writing $s - \rho = |s-\rho|e^{i\theta_\rho}$,
$\partial^2_{\sigma}\mathcal{L} = -\sum_\rho \cos(2\theta_\rho)/|s-\rho|^2$.
A term is positive when $\pi/4<|\theta_\rho|<3\pi/4$ (the zero lies in the vertical double
cone at $s$) and negative when $|\theta_\rho|<\pi/4$ or $|\theta_\rho|>3\pi/4$.
\end{remark}

\subsection{Sign on the critical line under RH}

\begin{proposition}\label{prop:sigma-convex}
Assume RH. Then for all $\gamma \notin \{\gamma_\rho\}$,
\[
  \frac{\partial^2}{\partial\sigma^2}\,\mathcal{L}\!\left(\tfrac{1}{2}+i\gamma\right) > 0.
\]
That is, $\sigma \mapsto \log|\xi(\sigma+i\gamma)|$ is strictly convex in $\sigma$
at $\sigma = 1/2$.
\end{proposition}

\begin{proof}
Under RH, $\beta_\rho = 1/2$ for all $\rho$, so each term in~\eqref{eq:curvature-cartesian}
at $\sigma=1/2$ equals $1/(\gamma-\gamma_\rho)^2 > 0$.
\end{proof}

\begin{remark}[Direction of convexity]
Under RH, $\log|\xi|$ is convex in $\sigma$ and, by harmonicity, concave in $t$ at
$\sigma = 1/2$.
\end{remark}

%------------------------------------------------------------------
\section{Positivity of \texorpdfstring{$\mathcal C$}{C} and the Laguerre--P\'olya class}
\label{sec:lp}
%------------------------------------------------------------------

\subsection{The function \texorpdfstring{$f(t) = \xi(1/2+it)$}{f(t)=xi(1/2+it)} is real-valued}

\begin{proposition}\label{prop:real}
For all $t \in \mathbb{R}$, $\xi(1/2+it) \in \mathbb{R}$.
\end{proposition}

\begin{proof}
Combining the functional equation $\xi(s) = \xi(1-s)$ with
$\overline{\xi(s)} = \xi(\bar{s})$:
$\overline{\xi(1/2+it)} = \xi(1/2-it) = \xi(1-(1/2+it)) = \xi(1/2+it)$.
\end{proof}

\subsection{The Laguerre--P\'olya class}

\begin{definition}
An entire function $f$ belongs to the \emph{Laguerre--P\'olya class}~$\LP$ if it is a
uniform limit on compact sets of real polynomials with only real zeros, or equivalently if
it has the form
\[
  f(t) = c\,t^m e^{-\alpha t^2+\beta t}\prod_k\!\left(1-\frac{t}{t_k}\right)e^{t/t_k},
\]
with $c,\beta,t_k \in \mathbb{R}$, $\alpha \geq 0$, $\sum_k t_k^{-2} < \infty$.
\end{definition}

In particular,
a real entire function of genus at most one with only real zeros belongs to $\LP$. The
function $f$ is real on $\mathbb R$ (Proposition~\ref{prop:real}) and of order one, so the
following holds (see de~Bruijn~\cite{deBruijn1950}, Newman~\cite{Newman1976}):

\begin{proposition}\label{prop:rh-lp}
The following are equivalent:
\begin{enumerate}
  \item[\rm(1)] RH: every non-trivial zero of $\zeta$ satisfies $\Re(\rho) = 1/2$.
  \item[\rm(2)] $f \in\LP$.
\end{enumerate}
\end{proposition}

\begin{proof}
The zeros of $f$ are $t=-i(\rho-\tfrac12)$, which are real exactly when $\Re\rho=\tfrac12$.
If RH holds, $f$ is real entire of order one (hence genus at most one) with real zeros, so
$f\in\LP$. Conversely, functions in $\LP$ have only real zeros.
\end{proof}

\subsection{Positivity of the curvature}

For $\gamma\in\mathbb R$ with $f(\gamma)\neq0$ we have, by harmonicity of $\mathcal L$,
\[
  \mathcal{C}(\gamma) = \partial^2_{\sigma}\mathcal{L}(\tfrac12+i\gamma)
  = -\frac{d^2}{dt^2}\log|f(t)|\Big|_{t=\gamma}.
\]
This is the only definition of $\mathcal C$ used below.

\begin{theorem}\label{thm:rh-curvature}
If RH holds, then $\mathcal C(\gamma)=\sum_\rho(\gamma-\gamma_\rho)^{-2}>0$ for every
$\gamma\notin\{\gamma_\rho\}$. Conversely, let $\rho_0=\tfrac12+b_0+i\gamma_0$ with
$b_0\neq0$ be a zero, and assume $\xi(\tfrac12+i\gamma_0)\neq0$. Its contribution to
$\mathcal C(\gamma_0)$, counted together with the zero $1-\bar\rho_0$, is $-2/b_0^2$, and
$\mathcal C(\gamma_0)<0$ whenever
\[
  \sum_{\rho\neq\rho_0,\,1-\bar\rho_0}\Re\frac{-1}{(\tfrac12+i\gamma_0-\rho)^2}<\frac{2}{b_0^2},
\]
where the sum removes one copy of each of $\rho_0$ and $1-\bar\rho_0$.
\end{theorem}

\begin{proof}
The first statement is Proposition~\ref{prop:sigma-convex}. For the second, put
$s=\tfrac12+i\gamma_0$. Then $s-\rho_0=-b_0$ and $s-(1-\bar\rho_0)=b_0$, so each of these two
zeros contributes $\Re(-1/b_0^2)=-1/b_0^2$ to~\eqref{eq:curvature-cartesian}. The total is
$-2/b_0^2$ plus the remaining sum, which is negative under the stated inequality.
\end{proof}

\begin{corollary}
If RH holds, then $\mathcal{C}(t) = \displaystyle\sum_\rho \frac{1}{(t-\gamma_\rho)^2}$
for all $t \notin \{\gamma_\rho\}$. Conversely, if this identity holds for all such $t$,
then RH holds.
\end{corollary}

\begin{proof}
The first statement is Theorem~\ref{thm:curvature} with $\beta_\rho=\tfrac12$. Conversely,
suppose the identity holds and let $\rho_0=\tfrac12+b_0+i\gamma_0$ with $b_0\ne0$ be a zero;
the right-hand side converges since $\sum_\rho\gamma_\rho^{-2}<\infty$. Let $k\ge0$ be the
number of zeros $\tfrac12+i\gamma_0$ and $k'\ge2$ the number of zeros with ordinate $\gamma_0$
off the line ($\rho_0$ and $1-\bar\rho_0$ among them). By~\eqref{eq:curvature-cartesian},
$(t-\gamma_0)^2\mathcal C(t)\to k$ as $t\to\gamma_0$, since the off-line zeros give bounded
terms near $t=\gamma_0$; while $(t-\gamma_0)^2\sum_\rho(t-\gamma_\rho)^{-2}\to k+k'$. This
contradicts the identity, so RH holds.
\end{proof}

\begin{remark}\label{rem:counterexample}
Whether $\mathcal C>0$ on $\mathbb R\setminus\{\gamma_\rho\}$ implies RH is open. The
negative contribution of an off-line pair can be masked by the positive contributions
$(\gamma-\gamma_\rho)^{-2}$ of nearby zeros on the line, whose number grows like
$(\log T)/2\pi$ per unit length. The analogous statement for general real entire functions
is false: $g(t)=(t^2+1)\sin\pi t$ is real entire of order one with the non-real zeros $\pm i$,
yet
\[
  -\frac{d^2}{dt^2}\log|g(t)|=\frac{\pi^2}{\sin^2\pi t}-\frac{2(1-t^2)}{(1+t^2)^2}
  \ge\pi^2-2>0\qquad(t\notin\mathbb Z).
\]
\end{remark}

\section{Related formulations}
\label{sec:fronts}
%------------------------------------------------------------------

We compare the positivity of $\mathcal C$ with several known formulations of RH.

\subsection{The de Bruijn--Newman constant}

Let
\[
  \Phi(u)=\sum_{n\ge1}(2\pi^2n^4e^{9u}-3\pi n^2e^{5u})e^{-\pi n^2e^{4u}}
\]
and $H_t(z)=\int_0^\infty e^{tu^2}\Phi(u)\cos(zu)\,du$, so that
$H_0(z)=\tfrac18\xi(\tfrac12+\tfrac{iz}{2})$. Let
\[
  \Lambda := \inf\{t \in \mathbb{R} : H_t \text{ has only real zeros}\}.
\]
De~Bruijn~\cite{deBruijn1950} and Newman~\cite{Newman1976} showed that $\Lambda$ is finite
and that $\mathrm{RH}\iff\Lambda\le0$; with $\Lambda\ge0$ \cite{RodgersTao2020}, RH holds if
and only if $\Lambda=0$. Since $H_0(z)=\tfrac18 f(z/2)$, the condition $\Lambda\le 0$ is
equivalent to $f\in\LP$.

\subsection{The negative index of the Weil form}

Let $Q$ be the Weil quadratic form on the test space $\mathcal S$ of \cite[Def.~2.1]{P25}
and $\kappa(Q)$ its negative index \cite[Def.~2.3]{P25}.

\begin{remark}\label{rem:kappa}
By Weil's positivity theorem \cite[Thm.~2.2]{P25}, $\mathrm{RH}\iff\kappa(Q)=0$. If $\zeta$
has finitely many zeros off the critical line, forming $m\ge1$ orbits under
$\rho\mapsto\bar\rho,\ \rho\mapsto1-\rho$ (Hypothesis~D of \cite[Def.~2.4]{P25}), then
$\kappa(Q)=2m$ \cite[Prop.~4.7]{P25}. Without the finiteness hypothesis no formula for
$\kappa(Q)$ in terms of the off-line zeros is proved there. A spectral-flow interpretation of
$\kappa(Q)$, in the spirit of Atiyah--Patodi--Singer~\cite{APS1975}, would require a family
of self-adjoint operators whose spectral flow equals $\kappa(Q)$; we do not construct one.
\end{remark}

\subsection{\texorpdfstring{$K$}{K}-theory}

\begin{remark}\label{rem:ktheory}
A $K$-theoretic formulation in the spirit of Connes~\cite{Connes1999} would require a class
in $K_0$ detecting $\kappa(Q)$; we do not construct one. Note that the naive candidate does
not work: $C_\mathbb{Q}\cong\mathbb R_+^*\times\hat{\mathbb Z}^*$, so
$C^*(C_\mathbb Q)\cong C_0(\mathbb R\times D)$ with $D$ the countable discrete dual of
$\hat{\mathbb Z}^*$, and $K_0(C^*(C_\mathbb Q))\cong\bigoplus_D K^0(\mathbb R)=0$.
\end{remark}

\subsection{Transversal energy}

\begin{remark}\label{rem:energy}
One may ask whether RH can be phrased as the minimality, in the transversal directions
$b_j=\Re\rho_j-\tfrac12$, of an energy of the zero configuration such as the transversal
energy $\mathcal T=\sum_j b_j^2$ plus logarithmic interaction terms. This is a heuristic: the
sums over infinitely many zeros diverge without renormalization, and we prove nothing about
it. Note also that $\int_{\mathbb R}\mathcal C(t)\,dt$ diverges under RH, since
$\mathcal C(t)\ge(t-\gamma_\rho)^{-2}$ near each ordinate.
\end{remark}

\subsection{Summary}

\begin{theorem}\label{thm:lp}
The following conditions are equivalent:
\begin{enumerate}
  \item[\rm(A)] $\Lambda = 0$ (de Bruijn--Newman).
  \item[\rm(B)] $\kappa(Q) = 0$ (negative index of the Weil form).
  \item[\rm(E)] $f \in \LP$.
  \item[\rm(F)] RH.
\end{enumerate}
Each of them implies
\begin{enumerate}
  \item[\rm(D)] $\mathcal{C}(t) > 0$ for all $t \notin \{\gamma_\rho\}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(F)$\iff$(E): Proposition~\ref{prop:rh-lp}.
(A)$\iff$(F): de~Bruijn--Newman and Rodgers--Tao, as recalled above.
(B)$\iff$(F): \cite[Thm.~2.2, Def.~2.3]{P25}.
(F)$\Rightarrow$(D): Theorem~\ref{thm:rh-curvature}.
\end{proof}

%------------------------------------------------------------------
\section{The sector decomposition of \texorpdfstring{$G(t)$}{G(t)}}
\label{sec:sectors}
%------------------------------------------------------------------

Define $G(t) := \Re\,(\xi'/\xi)'(1/2+it)$. By the Hadamard product,
\[
  G(t)=\sum_\rho\Re\frac{-1}{(1/2+it-\rho)^2}=-\frac{d^2}{dt^2}\log|f(t)|=\mathcal C(t).
\]
We decompose $G$ according to the factors of $\xi$.

\begin{theorem}[Sector decomposition]\label{thm:sectors}
For $t\in\mathbb R$ with $\zeta(\tfrac12+it)\neq0$,
\[
  G(t) = G_{\mathrm{poly}}(t) + G_\Gamma(t) + G_\zeta(t),
\]
where
\begin{align}
  G_{\mathrm{poly}}(t) &= \frac{2(t^2-1/4)}{(t^2+1/4)^2}, \label{eq:gpoly}\\[4pt]
  G_\Gamma(t) &= \tfrac{1}{4}\,\Re\!\left[\psi^{(1)}\!\left(\tfrac{1}{4}+\tfrac{it}{2}\right)\right],
  \label{eq:ggamma}\\[4pt]
  G_\zeta(t) &= -\frac{d^2}{dt^2}\log|\zeta\!\left(\tfrac{1}{2}+it\right)|.
  \label{eq:gzeta}
\end{align}
\end{theorem}

\begin{proof}
On $s=\tfrac12+it$,
$\log|f(t)| = \log|\tfrac{1}{2}s(s-1)| - \tfrac{\log\pi}{4} + \log|\Gamma(s/2)| + \log|\zeta(s)|$,
and $G=-\frac{d^2}{dt^2}\log|f|$.

\medskip\noindent\textit{Polynomial sector.} $|\tfrac{1}{2}s(s-1)| = \tfrac{1}{2}(t^2+1/4)$,
and $-\frac{d^2}{dt^2}\log(t^2+\tfrac14)=2(t^2-\tfrac14)/(t^2+\tfrac14)^2$.

\medskip\noindent\textit{Gamma sector.} $\frac{d^2}{dt^2}\log|\Gamma(1/4+it/2)| =
\Re[(i/2)^2\psi^{(1)}(1/4+it/2)] = -\tfrac{1}{4}\Re\psi^{(1)}(1/4+it/2)$, giving~\eqref{eq:ggamma}.
The $\zeta$ sector is~\eqref{eq:gzeta} by definition.
\end{proof}

\subsection{Properties of the analytic sector \texorpdfstring{$G_{\mathrm{ana}} = G_{\mathrm{poly}} + G_\Gamma$}{G\_ana}}

\begin{proposition}\label{prop:gana}
The analytic sector satisfies:
\begin{enumerate}
  \item[\rm(i)] $G_{\mathrm{poly}}(t) < 0$ for $|t| < 1/2$ and $G_{\mathrm{poly}}(t) > 0$ for
    $|t| > 1/2$; $G_{\mathrm{poly}}(t) = 2/t^2+O(t^{-4})$ as $t \to +\infty$.
  \item[\rm(ii)] $G_\Gamma(t) = -\tfrac1{4t^2}+O(t^{-4})$ as $t \to +\infty$.
  \item[\rm(iii)] For real $t$,
    \[
      \Re\!\left[\psi^{(1)}\!\left(\tfrac{1}{4}+\tfrac{it}{2}\right)\right]
      + \Re\!\left[\psi^{(1)}\!\left(\tfrac{3}{4}+\tfrac{it}{2}\right)\right]
      = 2\pi^2\operatorname{sech}^2(\pi t).
    \]
  \item[\rm(iv)] $G_{\mathrm{ana}}(0) = -8 + \tfrac{1}{4}\psi^{(1)}(1/4) = -8+\tfrac{\pi^2}4+2K
    \approx -3.70 < 0$, where $K$ is Catalan's constant.
  \item[\rm(v)] $G_{\mathrm{ana}}(t) = \tfrac{7}{4t^2}+O(t^{-4})$ as $t \to +\infty$.
\end{enumerate}
In particular, $G_{\mathrm{ana}}$ has at least one positive zero; let $t_*$ denote the
smallest one. Since $G_{\mathrm{ana}}$ is even, continuous and $G_{\mathrm{ana}}(0)<0$, we have
$G_{\mathrm{ana}}<0$ on $(-t_*,t_*)$. Numerically, $t_*\approx 0.4406$, and on $(0,50]$ this is the only zero.
\end{proposition}

\begin{proof}
(i) is immediate from~\eqref{eq:gpoly}.
(ii) Stirling's formula gives
$\psi^{(1)}(z)=z^{-1}+\tfrac12z^{-2}+\tfrac16z^{-3}+O(|z|^{-5})$ as $|z|\to\infty$ in
$|\arg z|<\pi-\delta$. For $z=\tfrac14+\tfrac{it}2$,
$\Re z^{-1}=\tfrac14/(\tfrac1{16}+\tfrac{t^2}4)=t^{-2}+O(t^{-4})$,
$\Re\tfrac12z^{-2}=-2t^{-2}+O(t^{-4})$ and $\Re z^{-3}=O(t^{-4})$, since
$z^{-3}=(it/2)^{-3}(1+O(t^{-1}))$ with $(it/2)^{-3}$ purely imaginary. Hence
$\Re\psi^{(1)}(z)=-t^{-2}+O(t^{-4})$ and $G_\Gamma(t)=-\tfrac1{4t^2}+O(t^{-4})$.
(iii) The reflection formula $\psi^{(1)}(z)+\psi^{(1)}(1-z)=\pi^2/\sin^2(\pi z)$ with
$z=\tfrac14+\tfrac{it}2$ gives, since $\Re\psi^{(1)}(\bar w)=\Re\psi^{(1)}(w)$, that the
left-hand side equals $\Re\,\pi^2/\sin^2(\tfrac\pi4+\tfrac{i\pi t}2)$. Now
$\sin^2(\tfrac\pi4+iy)=\tfrac12(1-\cos(\tfrac\pi2+2iy))=\tfrac12(1+i\sinh 2y)$, so
$\Re\,1/\sin^2(\tfrac\pi4+iy)=2/(1+\sinh^2 2y)=2\operatorname{sech}^2(2y)$; with
$y=\pi t/2$ this is the claim.
(iv) $G_{\mathrm{poly}}(0)=-8$ and $\psi^{(1)}(1/4)=\pi^2+8K\approx17.197$.
(v) Combine (i) and (ii): $2/t^2-1/(4t^2)=7/(4t^2)$.
The existence of a positive zero follows from (iv), (v) and continuity. The location and
uniqueness on $(0,50]$ are numerical evidence from evaluation of $t^2G_{\mathrm{ana}}(t)$ on a
grid; we do not prove uniqueness.
\end{proof}

\subsection{The arithmetic part}

\begin{remark}[Off-line zeros]\label{prop:wall}
An off-line zero $\rho_0=\tfrac12+b_0+i\gamma_0$ contributes $-2/b_0^2$ to
$\mathcal C(\gamma_0)=G(\gamma_0)$, together with $1-\bar\rho_0$
(Theorem~\ref{thm:rh-curvature}); this contribution is finite, and whether it makes $G$
negative depends on the other zeros. The positivity of $G=\mathcal C$ for $t$ outside the
ordinates is a consequence of RH; whether it implies RH is open. Since $G_{\mathrm{ana}}>0$
for large $t$ by Proposition~\ref{prop:gana}(v), positivity of $G$ for large $t$ would follow
from $G_\zeta(t)>-G_{\mathrm{ana}}(t)$; the term $G_\zeta$ is governed by the zeros of $\zeta$,
and the Dirichlet series $-(\zeta'/\zeta)(s)=\sum_n\Lambda_{\mathrm{vM}}(n)n^{-s}$ ($\Lambda_{\mathrm{vM}}$ the von
Mangoldt function) has abscissa of convergence $1$ and diverges for $\Re s<1$, so it gives no direct control of $G_\zeta$
on $\sigma=1/2$.
\end{remark}

\section{Summary of the decomposition}
\label{sec:wall}
%------------------------------------------------------------------

\begin{theorem}\label{thm:wall}
The following hold.
\begin{enumerate}
  \item[\rm(1)] RH is equivalent to $f \in \LP$ (Proposition~\ref{prop:rh-lp}).
  \item[\rm(2)] RH implies $G(t) = G_{\mathrm{ana}}(t) + G_\zeta(t) > 0$ for all
    $t \notin \{\gamma_\rho\}$.
  \item[\rm(3)] RH implies $G_\zeta(t) > -G_{\mathrm{ana}}(t)$ for all
    $t \notin \{\gamma_\rho\}$; for large $t$ the right-hand side is
    $-\tfrac{7}{4t^2}+O(t^{-4})$, and for $|t|<t_*$ it is positive.
\end{enumerate}
The analytic sector $G_{\mathrm{ana}}$ is explicit (Theorem~\ref{thm:sectors}).
Whether {\rm(2)} or {\rm(3)} implies RH is open.
\end{theorem}

\begin{proof}
(1) is Proposition~\ref{prop:rh-lp}; (2) is Theorem~\ref{thm:rh-curvature} with
$G=\mathcal C$; (3) is (2) rewritten with Theorem~\ref{thm:sectors} and
Proposition~\ref{prop:gana}.
\end{proof}

%------------------------------------------------------------------
\section{Conclusions}
\label{sec:conclusions}
%------------------------------------------------------------------

The Hadamard formula~\eqref{eq:curvature-cartesian} expresses
$\partial^2_{\sigma}\log|\xi|$ through the zeros; under RH it is positive on the critical line
outside the ordinates (Theorem~\ref{thm:rh-curvature}). The converse is open, and its analogue
for general real entire functions of order one is false
(Remark~\ref{rem:counterexample}). The positivity is implied by each of the equivalent
conditions $\Lambda=0$, $\kappa(Q)=0$, $f\in\LP$ (Theorem~\ref{thm:lp}).

The sector decomposition (Theorem~\ref{thm:sectors}) shows that the analytic part
$G_{\mathrm{ana}} = G_{\mathrm{poly}} + G_\Gamma$ is negative on $(-t_*,t_*)$, where $t_*$ is its smallest positive zero
(numerically $t_*\approx0.44$), and is asymptotic to $7/(4t^2)$; near $t=0$ the arithmetic
part $G_\zeta$ must compensate whenever $G>0$.

\medskip

\noindent\textbf{What is known.}
The formula~\eqref{eq:curvature-cartesian} follows directly from the Hadamard product; a
related positivity criterion for $\Re(\xi'/\xi)$ is due to Lagarias~\cite{Lagarias1999}.
The equivalences $\mathrm{RH}\iff f\in\LP\iff\Lambda\le0$ are due to
de~Bruijn~\cite{deBruijn1950} and Newman~\cite{Newman1976}; $\Lambda\ge0$ is
Rodgers--Tao~\cite{RodgersTao2020}; $\mathrm{RH}\iff\kappa(Q)=0$ is Weil's theorem
\cite[Thm.~2.2]{P25}. The explicit polygamma formulas of
Theorem~\ref{thm:sectors} and Proposition~\ref{prop:gana} are elementary computations.
