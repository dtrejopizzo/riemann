\title[The Band-Limited Weil--Carleson Form]{The Band-Limited Weil--Carleson Form: a numerically saturated generalized eigenvalue, sine-kernel structure of its spectrum, and a comparison of five reformulations}
\author{David Alejandro Trejo Pizzo}
\thanks{Independent Researcher. \texttt{dtrejopizzo@gmail.com}}


\begin{document}
\begin{abstract}
For test functions of exponential type $d$ the prime side of Weil's explicit formula is a finite sum over
$n\le e^{2d}$. We study numerically a generalized eigenvalue of the prime Gram matrix relative to the
archimedean Gram matrix on a sinc basis at height $T_0$, observe that it equals $1$ to five digits at the
computed values $2d/\log T_0=0.87$ and $1.45$ at $T_0=1000$, observe a sine-kernel structure in the subdominant spectrum, and
discuss heuristic relations with the Connes--Consani prolate approach.
\end{abstract}

\section{Introduction}\label{sec:intro}
Weil's criterion \cite{Weil1952,Bombieri2000} states that the Riemann Hypothesis is equivalent to the
positivity of the Weil quadratic functional, the distribution appearing in the explicit formula paired
against test functions of positive type. Earlier papers of the series \cite{Pdetector,Pkrein,Pstable,Panatomy} studied localized versions numerically. The present
paper studies a finite, computable version on band-limited test functions, which we call the
\emph{band-limited Weil--Carleson form}: the associated constant~\eqref{eq:Cdef} is the best
constant in an embedding inequality $\int|f|^2F\le \Cband\int|f|^2\Phi$, of the type that defines Carleson
measures, which explains the name.

We record three things. First, an exact statement: for test functions of exponential type $d$ only
$n\le e^{2d}$ enter the prime side (Proposition~\ref{prop:trunc}). Second, numerical observations at fixed
height: the generalized eigenvalue $\Cband$ equals $1$ to five digits at the computed values $2d/\log T_0=0.87$ and $1.45$
at $T_0=1000$ (Observation~\ref{thm:saturation}), and the subdominant spectrum is close to a sine-kernel Gram
matrix of the zeros (Observation~\ref{thm:sinekernel}). Third, a heuristic discussion comparing several
reformulations of the positivity, including the prolate approach of Connes and Consani
\cite{CC2021,CCprolate}.

\paragraph{Notation.} $\psi$ denotes the digamma function, $\Lambda$ the von Mangoldt function, and
$\gamma_\rho$ the (possibly complex) number with $\rho=\half+i\gamma_\rho$ for a nontrivial zero $\rho$.
The Fourier transform is $\hat\varphi(r)=\int_{\mathbb R}\varphi(u)e^{iru}\,du$, so that $f=\hat\varphi$ lives
on the frequency line $r$ and $\varphi$ on the logarithmic variable $u$. The normalized sinc is
$\operatorname{sinc}(x)=\sin(\pi x)/(\pi x)$. A \emph{window} at height $T_0$ of half-width $W$ is the
interval $[T_0-W,T_0+W]$. $\Phi$ denotes the archimedean density~\eqref{eq:PhiF}.

\section{The band-limited Weil--Carleson form}\label{sec:form}

\subsection{The exact finite reformulation}
Let $\varphi$ be of bounded variation with support in $[-d,d]$ and $f=\hat\varphi$. Then $f$ is entire of
exponential type $d$ and, integrating by parts against $d\varphi$, $|f(z)|\le e^{d|\mathrm{Im}\,z|}\,
\mathrm{Var}(\varphi)/|z|$, so $f(z)=O(1/|z|)$ on every strip $|\mathrm{Im}\,z|\le c$. This class contains the
sinc functions $S_j$ used below (for which $\varphi$ is a multiple of $e^{-ir_ju}\mathbf 1_{[-d,d]}$) and
$C_c^\infty(-d,d)$, which is what Weil's criterion requires. The test of positive type is
$h(z)=f(z)\overline{f(\bar z)}$, which equals $|f|^2$ on $\mathbb R$ and is of exponential type $2d$. Put
\begin{equation}\label{eq:PhiF}
\Phi(r)=\re\,\psi\!\left(\tfrac14+\tfrac{ir}{2}\right)-\log\pi,\qquad
F(r)=2\sum_{n\ge 2}\frac{\Lambda(n)}{\sqrt n}\cos(r\log n).
\end{equation}

\begin{proposition}\label{prop:trunc}
For $f$ as above,
\begin{equation}\label{eq:Wphi}
\begin{aligned}W(\varphi)\;:=\;&h(\tfrac i2)+h(-\tfrac i2)\;+\;\frac1{2\pi}\int_{\mathbb R}|f(r)|^2\,\Phi(r)\,dr\\
&\;-\;\frac1{2\pi}\int_{\mathbb R}|f(r)|^2\,F(r)\,dr \;=\;\sum_{\rho}h(\gamma_\rho),\end{aligned}
\end{equation}
the sum running over all nontrivial zeros, and $h(\frac i2)+h(-\frac i2)=2\,\re\big(f(\tfrac i2)\overline{f(-\tfrac i2)}\big)$.
In the last integral only the terms with $n\le e^{2d}$ contribute, so it is a finite sum. RH holds if and
only if $W(\varphi)\ge0$ for all such $f$ and all $d$ \textup{(}Weil \cite{Weil1952}; see
\cite{Bombieri2000}\textup{)}.
\end{proposition}

\begin{proof}
The identity is the explicit formula for $\zeta$ \cite{Weil1952} applied to $h$, with the prime term
\[\textstyle\sum_n\Lambda(n)n^{-1/2}\big(g(\log n)+g(-\log n)\big),\qquad g(u)=\frac1{2\pi}\int h(r)e^{-iru}dr,\]
rewritten as $\frac1{2\pi}\int h F$. Here $h$ is entire with $h(z)=O(1/|z|^2)$ on $|\mathrm{Im}\,z|\le1$, and
$g$ is, up to a constant, the convolution of $\varphi$ with $u\mapsto\overline{\varphi(-u)}$, hence Lipschitz with
support in $[-2d,2d]$; these conditions suffice for the explicit formula in the form of Barner \cite{Barner}, and all terms converge absolutely
($\Phi(r)=O(\log|r|)$, and $\sum_\rho|h(\gamma_\rho)|<\infty$ since the number of zeros of height in $[T,T+1]$ is
$O(\log T)$). Only $\log n\le 2d$ contributes. The pole value follows from $h(\pm\frac i2)=f(\pm\frac
i2)\overline{f(\mp\frac i2)}$. The equivalence with RH is Weil's criterion.
\end{proof}

The pole term has no sign in general. It equals $2|f(\frac i2)|^2\ge0$ when $f$ is even, $f(-z)=f(z)$. The finiteness of the prime side is a statement on
the band-limited class; it is not a pointwise comparison of $\Phi$ with the distribution $F$.

\subsection{The constant}
Fix a height $T_0$. Discretize the band-limited class by the Shannon basis
$S_j(r)=\operatorname{sinc}\!\big(d(r-r_j)/\pi\big)$, $r_j=T_0+(j-\tfrac{N+1}2)\pi/d$, $j=1,\dots,N$, which spans
a window of half-width $W=\tfrac{N-1}{2}\pi/d$, and define
\begin{equation}\label{eq:grams}
\Aphi[i,j]=\int S_i S_j\,\Phi\,dr,\qquad \Pf[i,j]=\int S_i S_j\,F\,dr.
\end{equation}
One has $\Phi(r)<0$ for $|r|<6.2898\ldots$ and $\Phi>0$ beyond. In the computations below the window lies far
inside $|r|>6.29$ and $\Aphi$ is numerically positive definite (smallest eigenvalue between $3.2$ and $15.8$ at
$T_0=1000$, and $4.8$ at $T_0=80$). We define, as a numerical diagnostic,
\begin{equation}\label{eq:Cdef}
\boxed{\ \Cband(d,T_0;N)\;:=\;\lambda_{\max}\!\big(\Pf,\Aphi\big)\ }
\end{equation}
on the $N$-dimensional sinc span. Since the pole term is not included and the span is finite, $\Cband\le1$ is
neither implied by RH nor known to imply it; the relation with RH is Proposition~\ref{prop:trunc}, which
includes the pole term and the whole class. The factor $2$ on $F$ and the form of $\Phi$ are the constants of
the explicit formula, so no normalization is free.

\section{First order: numerical saturation}\label{sec:first}

\subsection{The zero-side identity}
By Proposition~\ref{prop:trunc}, on the full band-limited class
\begin{equation}\label{eq:zeroside}
\frac1{2\pi}\int|f|^2(\Phi-F)\,dr\;=\;\sum_{\rho}h(\gamma_\rho)\;-\;h(\tfrac i2)-h(-\tfrac i2)
\end{equation}
(all zeros). On a finite sinc span centred at $T_0$, the contributions of zeros outside the window are small
but nonzero, and the pole term is small for $T_0$ large; the matrix $\Aphi-\Pf$ is therefore close to, but
not exactly, a sum of $K$ rank-one evaluation forms at the $K$ zeros in the window. A vector $f$ in the span
that nearly vanishes at those zeros makes $\Aphi-\Pf$ nearly zero; this is the heuristic mechanism behind the
following observation.

\begin{observation}[Numerical saturation]\label{thm:saturation}
At $T_0=1000$ on a $29$-dimensional sinc basis, the computed $\lambda_{\max}(\Pf,\Aphi)$ equals $1$ to five
digits at the two computed values $d=3$ and $d=5$ \textup{(}$2d/\log T_0=0.87$ and $1.45$\textup{)} and is below $1$ at $d=1,2,2.5$; see the table.
\end{observation}

\noindent\emph{Numerical record} (window at $T_0=1000$, $\log T_0=6.908$; sinc basis, $N=29$):
\begin{center}
\begin{tabular}{rrrrr}
\toprule
$d$ & $x=2d/\log T_0$ & $\pi(e^{2d})$ & $\Cband(d,T_0)$ & $1-\Cband$\\
\midrule
1.0 & 0.290 & 4    & 0.339 & $0.661$\\
2.0 & 0.579 & 16   & 0.785 & $0.215$\\
2.5 & 0.724 & 34   & 0.922 & $0.078$\\
3.0 & 0.869 & 79   & 1.00000 & $0.00000$\\
5.0 & 1.448 & 2466 & 1.00000 & $0.00000$\\
\bottomrule
\end{tabular}
\end{center}
(The computed values for $d=3,5$ are $0.999995$ and $1.000000$.) That $1-\Cband$ depends on $x$ alone, and
that adjoining primes only drives $\Cband$ towards $1$, are heuristic readings of these five values at a single
height and are not established.

\begin{remark}
The observation is consistent with the heuristic that the rational independence of the prime frequencies does
not by itself keep the prime term below the archimedean one on this class: by~\eqref{eq:zeroside} the
quadratic form $\Aphi-\Pf$ is governed by the zeros, and the numerics show no margin at the computed values $x=0.87$ and $1.45$.
\end{remark}

\section{Second order: sine-kernel structure}\label{sec:second}

If $\Aphi-\Pf$ were exactly $\sum_{\rho\in\text{window}}v_\rho v_\rho^{\!\top}$ with $v_\rho[j]=S_j(\gamma_\rho)$,
it would have rank $K$, and its nonzero spectrum relative to $\Aphi$ would be that of the $K\times K$ Gram
matrix $G[i,j]=\langle v_{\rho_i},v_{\rho_j}\rangle_{\Aphi^{-1}}$. Since $\Phi$ varies slowly, $G$ is, after
diagonal normalisation, close to the reproducing kernel of the band-limited space on the zeros, the sine kernel
$\sin\!\big(d(\gamma_i-\gamma_j)\big)/\big(d(\gamma_i-\gamma_j)\big)$. By the discussion after~\eqref{eq:zeroside}
these statements hold only approximately.

\begin{observation}[Sine-kernel structure]\label{thm:sinekernel}
At $d=1.3$, $T_0=80$, $N=29$ \textup{(}window $[46.17,113.83]$, containing $K=27$ zeros\textup{)}, the
computed spectrum of $\Aphi-\Pf$ has $27$ eigenvalues of order one \textup{(}the smallest $1.2$\textup{)},
followed by $0.5$ and $0.04$; the diagonal-normalised zero Gram $G$ agrees with the sine kernel to
$7\times10^{-2}$ in maximum norm.
\end{observation}

\begin{remark}[Off-line displacement]
If two zeros with ordinate $\tau$ (a double zero) were replaced by the pair $\gamma=\tau\pm i\delta$, then, for
$f$ with real Taylor coefficients at $\tau$, the two terms $2f(\tau)^2$ of $\sum_\rho h(\gamma_\rho)$ become
$h(\tau+i\delta)+h(\tau-i\delta)=2\re f(\tau+i\delta)^2=2f(\tau)^2-2\delta^2\big(f'(\tau)^2+f(\tau)f''(\tau)\big)+O(\delta^4)$.
For $f(\tau)=0\ne f'(\tau)$ this is $-2\delta^2f'(\tau)^2+O(\delta^4)$, so the form detects the displacement at
order $\delta^2$, provided $f$ is also small at the remaining zeros. We do not quantify the constant.
\end{remark}

\begin{remark}
Montgomery's pair-correlation theorem \cite{Montgomery} (conditional on RH, for Fourier support in $(-1,1)$)
describes the distribution of real zeros; it does not imply their reality. Numerical evidence for sine-kernel
statistics is in \cite{Odlyzko}, and the random-matrix framework in \cite{RudnickSarnak}.
\end{remark}

\section{The adelic analogy: the Connes--Consani prolate}\label{sec:connes}

Connes' trace formula reads the explicit formula as a sum over places of local Weil distributions, and RH is
the positivity of the global functional \cite{Connes1999}. Connes and Consani proved the archimedean positivity
\cite{CC2021} and studied the prolate spheroidal operators in \cite{CCprolate}. The prolate operator $Q_c$ on
$L^2[-1,1]$ has kernel $\sin\!\big(c(x-y)\big)/\big(\pi(x-y)\big)$ and eigenvalues $\lambda_n(c)\in(0,1)$
\cite{SlepianPollak}.

\begin{observation}[Prolate analogy]\label{obs:prolate}
The integral operator with kernel $\sin(d(x-y))/(\pi(x-y))$ on an interval of half-width $W$ is unitarily
equivalent, by rescaling, to $Q_c$ with $c=Wd$, whose eigenvalues lie in $(0,1)$ \cite{SlepianPollak}. We do not claim a
relation between this fact and positivity of $\Aphi$, which is verified numerically in our cases. Numerically,
$\lambda_n(c)$ has about $2c/\pi$ eigenvalues near $1$:
\begin{center}
\begin{tabular}{rrrr}
\toprule
$c$ & $2c/\pi$ & $\lambda_{\max}(c)$ & \#$\{\lambda_n>0.999\}$\\
\midrule
5 & 3.18 & $0.99935$ & 1\\
10 & 6.37 & $1-4.4\times10^{-8}$ & 3\\
20 & 12.73 & $1-O(10^{-16})$ (below double precision) & 9\\
40 & 25.46 & below double precision & 21\\
\bottomrule
\end{tabular}
\end{center}
\end{observation}

The reason to pass to the multiplicative (adelic) organization is the hope that the finite places become
individually positive, where the additive prime term does not (the local prime form is indefinite,
\cite{Panatomy}). The local Weil distribution $W_p$ is the same prime-power object, and its positivity is
global, not per-place. In the Connes--Consani approach the remaining step is the uniformity of the cutoff
positivity as the adelic scale $L\to\infty$, which involves the transition modes $\lambda_n\approx\tfrac12$.
We regard this, heuristically, as analogous to the uniform-passage questions discussed in
\cite{Pdetector,Pkrein}.

\section{Five reformulations: a heuristic comparison}\label{sec:map}

We collect several reformulations of the positivity and indicate, for each, which part is known to be positive
and where the difficulty lies. Rows 3 and 4 concern the same matrix.

\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{p{0.5cm}p{4.5cm}p{4.3cm}p{3.6cm}}
\toprule
 & \textbf{Language} & \textbf{Known part} & \textbf{Difficulty}\\
\midrule
1 & Explicit formula / Kre\u{\i}n formulation \cite{Pkrein} & under a coercivity hypothesis, form represented by a bounded self-adjoint operator \cite{Pkrein} & the representation does not determine the sign\\
2 & Per-place (local) Weil form & archimedean place positive \cite{CC2021} & local prime form indefinite \cite{Panatomy}\\
3 & Band-limited constant (\S\ref{sec:first}) & $\Aphi\succ0$ numerically in the cases computed & $\Cband\approx1$ numerically; no visible margin\\
4 & Sine-kernel structure (\S\ref{sec:second}) & sine-kernel Gram matrix is positive semidefinite on real ordinates & presupposes real ordinates\\
5 & Adelic prolate (Connes--Consani, \S\ref{sec:connes}) & archimedean positivity \cite{CC2021} & finite places and uniformity as $L\to\infty$\\
\bottomrule
\end{tabular}
\end{center}

\begin{remark}[Discussion]\label{thm:wall}
In rows 2, 3 and 5 the known positive part is archimedean (in rows 2 and 5 by the theorem of \cite{CC2021}, which holds for test functions supported in a short interval; in row 3 numerically), and the difficulty sits
in the prime contribution. We do not know of a way, in any of these formulations, to control the prime
contribution other than through information equivalent to the location of the zeros. This is a heuristic
summary of the methods considered here, not a theorem about RH.
\end{remark}

\section{Conclusion}\label{sec:concl}

We have recorded the exact truncation of the prime side for test functions of exponential type $d$
(Proposition~\ref{prop:trunc}), the numerical observation that the generalized eigenvalue $\Cband$ equals $1$
to five digits at $T_0=1000$ at the computed values $2d/\log T_0=0.87$ and $1.45$, numerical sine-kernel structure in its subdominant
spectrum, and a heuristic comparison with the Connes--Consani prolate approach. On this heuristic reading, a method that would
decide the sign would have to control the prime contribution in a way that is not per-place, is uniform in the scale of the cutoff,
and does not presuppose the reality of the zeros.

\paragraph{Reproducibility.} The scripts \texttt{carleson\_engine.py}, \texttt{sinekernel.py} and
\texttt{prolate.py} in the paper folder reproduce every number (they use \texttt{numpy}, \texttt{scipy} and
\texttt{mpmath}; zeros of $\zeta$ are taken from \texttt{mpmath.zetazero}); their outputs are in
\texttt{results/}. The archimedean integrals are computed by the trapezoidal rule on $[T_0-R,T_0+R]$, with $R=400$ at
$T_0=1000$ and $R=79$ at $T_0=80$, and the tails beyond $T_0\pm R$ are added by adaptive quadrature, using
$S_iS_j=(-1)^{i-j}\sin^2(d(r-r_1))/(d^2(r-r_i)(r-r_j))$ split into a smooth and an oscillatory part; the prime
integrals are computed in closed form.
