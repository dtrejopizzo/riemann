\begin{abstract}
We directly attack $\varepsilon_0(\lambda)\ge0$ (localized Weil positivity
$\Leftrightarrow$ Link 2 $\Leftrightarrow$ RH below the horizon). The strategy,
suggested by route \#9, works on \emph{two levels}: \textbf{(I)} the kinetic--well
cancellation at \emph{leading order} is controlled by PNT (moments $M_0,M_2$,
coefficient $c=-\psi''(\tfrac14)/8$) --- \emph{unconditional}; \textbf{(II)} the
\emph{residue} after that cancellation is $\sim C_0/\lambda^2$ and its \emph{sign} is
governed by the rank-$1$ pole term (the pole of $\zeta$ at $s=1$,
unconditional). If the residue is dominated by the pole (Conjecture \ref{conj:main}),
then $\varepsilon_0\ge0$ \emph{without} using the position of the zeros. Route \#9
provides numerical support (rank-$1$ Hessian, pole direction). The single \GAP\ is a
scalar inequality between two quantities, \emph{both} computable on the arithmetic side.
\textbf{This is not a reformulation of RH: it is a concrete inequality whose proof
would give RH, with the key asymmetry that the pole is unconditional.}
\end{abstract}

\section{The form, in three pieces}\label{sec:pieces}
% =====================================================================

For $g\in\El=\mathrm{PW}_{L/2}$, $L=2\log\lambda$, Weil's explicit formula gives
\begin{equation}\label{eq:three}
  A_\lambda(g,g)
  \;=\;\kin(g)\;+\;\pole(g)\;-\;\vv_\lambda(g),
\end{equation}
with the three pieces \PR\ (established in the program):
\begin{itemize}[leftmargin=1.4em]
\item \textbf{Archimedean kinetic term} $\kin(g)\ge0$: jump form of the symbol
$\Omega(t)=-\log\pi+\Real\psi(\tfrac14+\tfrac{it}{2})$, conditionally negative
definite (Lévy--Khintchine, density $d\nu\ge0$), with kinetic coefficient
$c=-\tfrac18\psi''(\tfrac14)=16.166\ldots$ Asymptotically
$\kin(g)\gtrsim c\!\int|g'|^2$.
\item \textbf{Pole term} $\pole(g)$: contribution of the pole of $\zeta$ at $s=1$
(and $s=0$ by symmetry). It is a \emph{positive rank-$1$} operator:
\begin{equation}\label{eq:pole}
  \pole(g)\;=\;\kappa_\lambda\,\bigl|\langle g,\varphi_{\mathrm{pole}}\rangle\bigr|^2,
  \qquad \kappa_\lambda>0,
\end{equation}
where $\varphi_{\mathrm{pole}}$ is the uniform/global-scale direction identified
in route \#9. \emph{Crucially, $\kappa_\lambda$ and $\varphi_{\mathrm{pole}}$ are
unconditional}: the pole at $s=1$ does not depend on the position of any zero.
\item \textbf{Prime well} $\vv_\lambda(g)\ge0$: arithmetic jump form,
$\vv_\lambda(g)=\sum_{n\le\lambda^2}\frac{\Lam(n)}{\sqrt n}\,R_g(\log n)$, with
boundary moments $M_0\sim2\lambda^{1/2-|x|}$, $M_2\sim2\lambda^{1/2-|x|}(\log\lambda)^2$
(PNT, \PR).
\end{itemize}

\section{Level I: leading-order cancellation is PNT}\label{sec:leading}
% =====================================================================

\begin{proposition}[Unconditional leading balance]\label{prop:leading}
There exists a constant $c_\star$, computable from $M_0,M_2,c$ by a Hardy/uncertainty
inequality, such that for all $g\in\El$
\begin{equation}\label{eq:hardy}
  \kin(g)\;\ge\;c_\star\,\vv_\lambda(g)\;-\;o\!\left(\lambda^{-2}\right)\|g\|^2 .
\end{equation}
The \emph{exact} balance would require $c_\star=2$ (the tight constant); elementary
techniques give $c_\star<2$. \PR\ the leading-order balance: the dominant terms
$\sim M_0\sim\lambda^{1/2}$ of the kinetic term and the well cancel via PNT, leaving a
\textbf{residue} of order $\lambda^{-2}$.
\end{proposition}

\begin{remark}[Why the residue is $\lambda^{-2}$, not $\lambda^{1/2}$]
The \emph{raw} pieces grow like $\lambda^{1/2-|x|}$ (boundary moments), but they
cancel almost exactly: numerically $\varepsilon_0=A_\lambda(g_0,g_0)\sim
C_0/\lambda^2$, $35$--$40$ orders of magnitude below the individual blocks. The
leading-order cancellation \emph{is} the content of the archimedean$\leftrightarrow$primes
Weil balance; what PNT controls is this gross cancellation.
The \emph{sign} of what remains --- the residue $\sim\lambda^{-2}$ --- is where RH lives.
\end{remark}

\begin{definition}[The residue]
\[
  D_\lambda(g)\;:=\;(2-c_\star)\,\vv_\lambda(g)\;+\;[\text{corrections of order }\lambda^{-2}]
  \;\ge0,
\]
the \emph{deficit} by which the elementary kinetic term falls short of the exact balance.
Then $A_\lambda(g,g)=\bigl[\kin(g)-2\vv_\lambda(g)\bigr]+\pole(g)+D_\lambda(g)$,
and the problem $\varepsilon_0\ge0$ is equivalent to controlling the sign of
$\kin-2\vv+\pole+D$ on $\El$. The block $\kin-2\vv$ is the exact balance (PNT
leading order); the fine game is $\pole+D$ vs.\ the residue of $\kin-2\vv$.
\end{definition}

\section{Level II: the residue is rank--1 and aligned with the pole}\label{sec:residue}
% =====================================================================

Here enters the central result of route \#9.

\begin{proposition}[Structure of the residue, \NM\ via \#9]\label{prop:rank1}
The second variation of the residue $\varepsilon_k$ under perturbations of the prime
weights $\Lam(p)\mapsto\Lam(p)(1+\delta_p)$ is a Hessian of \textbf{rank $1$}
(mpmath, dps$=50$, $\lambda=3.7\ldots13$; second singular value $\le0.5\%$ of the first),
whose unique soft direction \textbf{converges} (angle $\to0.5^\circ$) to the uniform
vector $=\varphi_{\mathrm{pole}}$. That is: \emph{all the curvature of the residue
lives in the pole direction; the rest is kernel (zero curvature).}
\end{proposition}

\begin{remark}[The asymmetry that breaks the ``density vs.\ position'' symmetry]
The earlier verdict (\texttt{objective-B-verdict}) used $A_\lambda\ge0\Leftrightarrow$RH
to declare circularity. But it \emph{ignored the rank structure}: if the residue
$D_\lambda$ can only push $\varepsilon_0$ negative \emph{along
$\varphi_{\mathrm{pole}}$} (Prop.\ \ref{prop:rank1}), and exactly there the
pole term $\pole$ acts (positive, rank $1$, \emph{unconditional}), then the crossing to
negative could be \textbf{blocked by a term that does not depend on the zeros}.
This is the asymmetry that a generic positivity argument does not see.
\end{remark}

\section{The inequality that closes (Conjecture) and the \GAP}\label{sec:conj}
% =====================================================================

\begin{conjecture}[Pole $\succeq$ deficit domination]\label{conj:main}
With $\pole$ and $D_\lambda$ the two arithmetic quantities above, there exists
$\lambda_0$ such that for all $\lambda\ge\lambda_0$ and all $g\in\El$,
\begin{equation}\label{eq:main}
  \boxed{\;\bigl|\kin(g)-2\vv_\lambda(g)\bigr|_{-}\;\le\;\pole(g)\;+\;D_\lambda(g)\;}
\end{equation}
($|\cdot|_-$ the negative part). Equivalently, the smallest eigenvalue of the block
$\kin-2\vv$ restricted to $\El$ is $\ge-\kappa_\lambda$ along
$\varphi_{\mathrm{pole}}$ and $\ge0$ on the orthogonal complement.
\end{conjecture}

\begin{theorem}[Conditional]\label{thm:cond}
Conjecture \ref{conj:main} implies $\varepsilon_0(\lambda)\ge0$ for
$\lambda\ge\lambda_0$, and therefore (with unconditional E1 $+$ Hurwitz) \textbf{RH}.
\end{theorem}
\begin{proof}
$A_\lambda=(\kin-2\vv)+\pole+D_\lambda$. On $\varphi_{\mathrm{pole}}^\perp$, by
Prop.\ \ref{prop:rank1} the residue has no curvature, and $\kin-2\vv\ge0$ (leading
balance with no directional deficit); along $\varphi_{\mathrm{pole}}$, \eqref{eq:main}
gives $(\kin-2\vv)+\pole+D_\lambda\ge0$. Hence $A_\lambda\ge0$ on $\El$. \qed
\end{proof}

\paragraph{The \GAP, localized.}
\eqref{eq:main} is an inequality between \emph{three quantities on the arithmetic
side} ($\kin$, $\vv_\lambda$, $\pole$), \emph{all computable without the zeros}. What
is missing:
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item \GAP\ Prove that the directional deficit $|\kin-2\vv|_-$ is \emph{confined}
to $\varphi_{\mathrm{pole}}$ (Prop.\ \ref{prop:rank1} is \NM, still needs \PR).
Candidate: that the complement $\varphi_{\mathrm{pole}}^\perp$ inherits the CND
positivity of the archimedean symbol without a deficit (the Weil cancellation is
exact outside the pole).
\item \GAP\ Prove the scalar bound $\kappa_\lambda\ge\|\,|\kin-2\vv|_-\,\|$ along
the pole. Both sides are $\sim\lambda^{-2}$ (compatible magnitudes, unlike the
raw blocks $\sim\lambda^{1/2}$); the constant is still missing.
\end{enumerate}

\section{Why this is NOT a reformulation}\label{sec:notrefo}
% =====================================================================

\begin{enumerate}[leftmargin=1.6em]
\item A reformulation of RH would be ``$A_\lambda\ge0\Leftrightarrow$RH'', nothing
more. Here, instead, \eqref{eq:main} is a \textbf{structured} inequality: it splits
the problem into (a)~an \emph{unconditional} leading balance (PNT), (b)~an orthogonal
complement where positivity is \emph{automatic} (CND), and (c)~a \emph{single}
direction (pole) where all the risk is concentrated, controlled by an
\emph{unconditional} term (pole at $s=1$).
\item The decisive asymmetry: the term that must dominate the deficit is the
\textbf{pole}, which is \emph{unconditional}. An off-line zero would perturb the
deficit, but --- by the rank-$1$ structure --- only along the direction
where the pole already dominates. If this is formalized, off-line zeros are
\emph{excluded by a term that does not mention them}.
\item Therefore the RH content is not ``hidden in an equivalence'': it is
reduced to \textbf{two scalar arithmetic bounds} (i)--(ii), both $\sim\lambda^{-2}$,
both computable, with numerical support (route \#9). That is a proof program,
not a tautology.
\end{enumerate}

\paragraph{Honesty.} Props.\ \ref{prop:leading} and \ref{prop:rank1} are \PR/\NM;
Conjecture \ref{conj:main} and \GAP\ (i)--(ii) are what is missing. It is not proved.
But \textbf{it is not circular or vacuous}: it is a concrete inequality between
objects on the prime side, with a genuine asymmetry (unconditional pole) that the
earlier circularity argument overlooked. \emph{The battle continues here.}
