\begin{abstract}
\GAP\ (a) was the \emph{only} genuinely open piece of condition (i): proving
$\kin-\vv\succeq0$ on $\fp^\perp$ for the \emph{non--local} kinetic operator $\kin$
(CND symbol $\Omega$) and the prime well $\vv$. We show that the
\textbf{ground--state representation} (ground--state substitution) of
Frank--Lieb--Seiringer, valid for jump generators with kernel $\ge0$, gives an
\emph{exact} identity:
\[
  \kin(g)-\vv_0(g)=\tfrac12\!\iint |h(x)-h(y)|^2\,\fp(x)\fp(y)\,k(x,y)\,dx\,dy\;\ge0,
  \qquad g=\fp\,h,
\]
with manifestly positive, \emph{strict} remainder on $\fp^\perp$ ($h$ non-constant). Thus
the improved non--local Hardy inequality \textbf{holds} for the ground--state potential
$\vv_0:=\kin\fp/\fp$. The residual \GAP\ reduces to a \textbf{single arithmetic quantity}:
the \textbf{discrepancy} $\delta\vv:=\vv_\lambda-\vv_0$ between the true prime well and the
pole potential. \textbf{$\delta\vv$ is on the prime side (PNT), not the zero side}: it is
the only thing left to close (i).
\end{abstract}

\section{The kinetic term as a non--local Dirichlet form}\label{sec:dirichlet}
% =====================================================================

The archimedean symbol $\Omega(t)-\Omega(0)\ge0$ is CND with Lévy representation
\[
  \Omega(t)-\Omega(0)=\int_0^\infty\!\bigl(1-\cos(t\xi)\bigr)\,d\nu(\xi),
  \qquad d\nu(\xi)=\frac{2e^{-\xi/2}}{1-e^{-2\xi}}\,d\xi\ge0 .
\]
By Plancherel, the kinetic form is a \textbf{non--local Dirichlet form} with jump
kernel $k\ge0$:
\begin{equation}\label{eq:Kdirichlet}
  \kin(g)=\frac1{2\pi}\!\int|\widehat g(t)|^2(\Omega(t)-\Omega(0))\,dt
  =\tfrac12\!\iint_{\R^2}|g(x)-g(y)|^2\,k(x-y)\,dx\,dy,
\end{equation}
with $k(u)=\int_0^\infty (\text{cosine kernel})\,d\nu$, $k\ge0$. \PR\ (Beurling--Deny:
every CND symbol gives a jump form with $k\ge0$).

\section{The ground-state identity}\label{sec:gsr}
% =====================================================================

\begin{lemma}[Ground-state representation, Frank--Lieb--Seiringer]
\label{lem:gsr}
Let $k\ge0$ be a symmetric jump kernel, $\kin$ the Dirichlet form
\eqref{eq:Kdirichlet}, and $\omega>0$ a function with associated potential
$\vv_0:=\kin\omega/\omega$ (i.e.\ $\omega$ is an eigenfunction of $\kin$ with ``potential''
$\vv_0$ at energy $0$). Then, for $g=\omega h$,
\begin{equation}\label{eq:gsridentity}
  \kin(g)-\int \vv_0\,|g|^2
  \;=\;\tfrac12\!\iint_{\R^2} |h(x)-h(y)|^2\,\omega(x)\,\omega(y)\,k(x-y)\,dx\,dy
  \;\ge\;0 .
\end{equation}
\end{lemma}
\begin{proof}
Direct expansion: with $g=\omega h$,
$|g(x)-g(y)|^2=|\omega(x)h(x)-\omega(y)h(y)|^2$. Using
$|\omega_x h_x-\omega_y h_y|^2=\omega_x\omega_y|h_x-h_y|^2+(h_x^2\omega_x(\omega_x-\omega_y)+h_y^2\omega_y(\omega_y-\omega_x))$
and the symmetry of $k$, the second group reassembles into
$\int |h|^2\,\omega\,(\kin\omega)=\int\vv_0|g|^2$, leaving \eqref{eq:gsridentity}. This is the
Frank--Lieb--Seiringer (2008) identity for jump forms. \qed
\end{proof}

\begin{corollary}[Improved non--local Hardy inequality]\label{cor:improved}
Taking $\omega=\fp$ (the pole direction $=$ ground state, via \#9):
\[
  \kin(g)-\vv_0(g)\ge0\quad\forall g,\qquad
  \kin(g)-\vv_0(g)>0\quad\forall g\in\fp^\perp,\ g\ne0,
\]
where $\vv_0=\kin\fp/\fp$. Strictness on $\fp^\perp$: $g\perp\fp$ is equivalent to
$\int\fp^2 h=0$ ($h=g/\fp$ has zero mean in $\fp^2\,dx$), so $h$ is not constant and
the remainder \eqref{eq:gsridentity} is $>0$.
\end{corollary}

\noindent\textbf{This resolves \GAP\ (a) for the potential $\vv_0$}: the improved
non--local Hardy inequality \emph{holds}, unconditionally, via the ground-state
substitution. The non--locality --- the historical obstacle --- is \emph{absorbed}
in the identity \eqref{eq:gsridentity}, which is exact for jump kernels.

\section{The residual \GAP: the discrepancy $\delta\vv$}\label{sec:discrepancy}
% =====================================================================

The real well is $\vv_\lambda$ (primes), not $\vv_0=\kin\fp/\fp$. We write
\begin{equation}\label{eq:dV}
  \vv_\lambda=\vv_0+\delta\vv,\qquad \delta\vv:=\vv_\lambda-\frac{\kin\fp}{\fp}.
\end{equation}
By Corollary \ref{cor:improved}, on $\fp^\perp$,
\[
  \kin(g)-\vv_\lambda(g)
  =\underbrace{\bigl[\kin(g)-\vv_0(g)\bigr]}_{\ge\,\beta\|g\|^2\ \text{(remainder)}}
  -\int\delta\vv\,|g|^2 .
\]

\begin{proposition}[Condition that closes (i)]\label{prop:close}
If $\displaystyle \int\delta\vv\,|g|^2\le \beta_\lambda\|g\|^2$ for all
$g\in\fp^\perp$ (with $\beta_\lambda$ the gap from \eqref{eq:gsridentity}), then
$\kin-\vv_\lambda\succeq0$ on $\fp^\perp$, i.e.\ \GAP\ (i) is \PR.
\end{proposition}

\paragraph{Why $\delta\vv$ is PNT and not zeros.} Both $\vv_\lambda$ (von Mangoldt sum
$\le\lambda^2$) and $\vv_0=\kin\fp/\fp$ (archimedean kinetic term applied to the
pole direction) are quantities \emph{on the arithmetic side}: they do not involve the
position of any zero. The discrepancy $\delta\vv$ measures how much the prime well
deviates from the potential that would make $\fp$ the exact ground state.

\begin{remark}[Interpretation: $\delta\vv$ and the pole]
$\fp$ is the uniform/global-density direction. $\vv_0=\kin\fp/\fp$ is the ``smooth''
potential that the pole at $s=1$ generates. $\delta\vv=\vv_\lambda-\vv_0$ is the
\emph{oscillatory} part of the prime well (the prime-to-prime fluctuations around the
mean density). That $\delta\vv$ be small on $\fp^\perp$ is, essentially,
equidistribution of primes (PNT with error term) projected onto the complement of the
mean mode. \textbf{Candidate bound:} $\|\delta\vv\|_{\fp^\perp}\lesssim
(\log\lambda)^{-A}$ or a margin $\sim\lambda^{-2}$, via the prime number theorem with
classical zero-free regions (Vinogradov--Korobov) --- \emph{not} RH.
\end{remark}

\section{Verdict}\label{sec:verdict}
% =====================================================================

\begin{enumerate}[leftmargin=1.6em]
\item \PR\ The improved \emph{non--local} Hardy inequality for the ground-state
potential $\vv_0$: the Frank--Lieb--Seiringer substitution gives an exact identity
with remainder $\ge0$, strict on $\fp^\perp$. \textbf{Non--locality no longer
obstructs.}
\item \OP\ The complete \GAP\ (i) is reduced to a \emph{single} bound:
$\int\delta\vv\,|g|^2\le\beta_\lambda\|g\|^2$ on $\fp^\perp$, with
$\delta\vv=\vv_\lambda-\kin\fp/\fp$ a quantity \emph{on the prime side}.
\item \textbf{Reading.} The cascade of Link 2 / $\varepsilon_0\ge0$ has been reduced,
step by step and with classical tools (Schur, Frank--Lieb--Seiringer), to
\emph{two} arithmetic bounds: the discrepancy $\delta\vv$ (\S\ref{sec:discrepancy}) and
the Schur scalar (ii). Neither mentions the position of the zeros. Both are
candidates for control via PNT/classical zero-free regions.
\end{enumerate}

\noindent\textbf{Honesty.} The GSR identity \eqref{eq:gsridentity} is a classical
theorem correctly applied; it resolves the non--locality. But the reduction to
$\delta\vv$ \emph{transfers} the problem, it does not eliminate it: the discrepancy
bound still needs to be proved, and it needs to be verified that it is genuinely PNT
(density) and not, once again, position in disguise. That is the point to audit next.
The real progress: the historical obstacle (non--locality, $85\%$ large jumps)
\textbf{has stopped being an obstacle}.
