\title{\textbf{Attack on \GAP\ (i): the Hardy deficit lives in the pole mode.\\

\begin{abstract}
\GAP\ (i) was: proving that the kinetic--well deficit is \emph{confined} to the
pole direction $\fp$, i.e.\ $\kin-\vv\succeq0$ on $\fp^\perp$. We show
(\S\ref{sec:schur}) that, via the Schur complement for the rank-$1$ perturbation
$\pole=\kappa|\fp\rangle\langle\fp|$, positivity $A_\lambda\succeq0$ is \emph{exactly
equivalent} to two conditions: \textbf{(i)} $\kin-\vv\succeq0$ on $\fp^\perp$ and
\textbf{(ii)} a scalar Schur inequality. Condition (i) is an \textbf{improved Hardy
inequality}: the Hardy extremal (where the kinetic bound is tight, $c=2$) is
\emph{precisely} the uniform/constant mode $=\fp$; projecting it out, a \emph{positive
remainder} appears (Brezis--Vázquez), and the kinetic--well inequality becomes strict.
Thus \GAP\ (i) is \textbf{not} RH-equivalent: it is the non--local version of an
unconditional Hardy improvement. The residual \GAP\ is the non--locality of the
archimedean symbol (Lévy/CND).
\end{abstract}

\section{The Schur reduction (exact)}\label{sec:schur}
% =====================================================================

Let $B:=\kin-\vv$ (kinetic minus well, symmetric, on the arithmetic side) and
$\pole=\kappa_\lambda|\fp\rangle\langle\fp|$ the pole term (rank $1$, positive,
unconditional). The Weil form is $A_\lambda=B+\pole$. In the decomposition
$\El=\R\fp\oplus\fp^\perp$ we write in blocks
\[
  B=\begin{pmatrix} b & r^* \\ r & M\end{pmatrix},
  \qquad
  b=\langle\fp,B\fp\rangle\in\R,\quad r=\Pi_\perp B\fp,\quad M=\Pi_\perp B\,\Pi_\perp,
\]
where $\Pi_\perp$ is the projection onto $\fp^\perp$. Then
$A_\lambda=\begin{pmatrix} b+\kappa_\lambda & r^*\\ r & M\end{pmatrix}$.

\begin{lemma}[Schur reduction]\label{lem:schur}
$A_\lambda\succeq0$ \emph{if and only if}
\begin{enumerate}[label=\textbf{(\roman*)},leftmargin=2.2em]
\item $M\succeq0$\quad($\kin-\vv$ is positive semidefinite on $\fp^\perp$);
\item $r\in\ran(M)$\ \ and\ \ $b+\kappa_\lambda\;\ge\;r^*M^{+}r$\quad(scalar Schur
condition, $M^+$ pseudoinverse).
\end{enumerate}
\end{lemma}
\begin{proof}
Schur criterion for symmetric block matrices with positive semidefinite block
$(2,2)=M$. \qed
\end{proof}

\noindent This \emph{formalizes} the separation that route \#9 suggested: all of Weil
positivity splits into \textbf{(i)} positivity on the complement of the pole,
and \textbf{(ii)} a single scalar inequality along the pole. \GAP\ (i) is
exactly condition (i).

\section{\GAP\ (i) is an improved Hardy inequality}\label{sec:hardy}
% =====================================================================

Condition (i), $M\succeq0$, is
\begin{equation}\label{eq:gapi}
  \kin(g)\;\ge\;\vv_\lambda(g)\qquad\text{for all }g\in\El\cap\fp^\perp .
\end{equation}

\paragraph{Why the unrestricted version fails.}
The inequality \emph{without} projecting, $\kin(g)\ge\vv_\lambda(g)\ \forall g\in\El$, is
the Weil balance with tight constant $c=2$, which elementary techniques \emph{do not}
reach (they give $c<2$; the tight value $c=2$ is RH; cf.\
\texttt{A\_intento\_demostracion}). The deficit $c<2$ is real.

\begin{proposition}[The deficit is the Hardy extremal $=\fp$]\label{prop:extremal}
The underlying Hardy-type inequality,
$\kin(g)=\frac1{2\pi}\int|\widehat g|^2(\Omega(t)-\Omega(0))\,dt\ \gtrsim\
c\int|g'|^2$, has as its \emph{extremal} function (where the bound is tight) the
\emph{zero-frequency} / uniform mode. In the modal basis of $A_\lambda$, that extremal
is the ground state, which by route \#9 is precisely $\fp$ (the soft direction,
$=$ global density $=$ pole). \NM\ (via \#9: the curvature of the deficit is rank $1$
along $\fp$).
\end{proposition}

\begin{remark}[Hardy improvement: the Brezis--Vázquez remainder]
It is a classical phenomenon that Hardy inequalities become \emph{strict with
positive remainder} when restricted to the orthogonal complement of the extremal:
\[
  \int|g'|^2-c\!\int\!\frac{|g|^2}{\text{(weight)}}
  \;\ge\;\beta\int|g|^2_{\text{(remainder)}}>0
  \qquad\text{for }g\perp(\text{extremal}).
\]
If the extremal is $\fp$ (Prop.\ \ref{prop:extremal}), then on $\fp^\perp$ the
deficit $c<2$ \emph{closes}: the bound becomes $c=2$ with margin, giving
\eqref{eq:gapi}. \textbf{\GAP\ (i) is, therefore, a Hardy improvement, not RH.}
\end{remark}

\begin{conjecture}[Improved non--local Hardy inequality]\label{conj:hardy}
For the non--local kinetic operator $\kin$ (CND symbol $\Omega$, Lévy measure
$d\nu(\xi)=2e^{-\xi/2}(1-e^{-2\xi})^{-1}d\xi$) and the well $\vv_\lambda$ (PNT moments
$M_0,M_2$), there exists $\beta_\lambda\ge0$ such that
\[
  \kin(g)-\vv_\lambda(g)\;\ge\;\beta_\lambda\,\|g\|^2
  \qquad\text{for all }g\in\El\cap\fp^\perp .
\]
\end{conjecture}

\section{Counting bound states: why ``at most one''}\label{sec:bargmann}
% =====================================================================

Conjecture \ref{conj:hardy} is equivalent to the non--local ``Schrödinger'' operator
$\kin-\vv_\lambda$ having \textbf{at most one bound state} (one negative eigenvalue), and
its eigenvector being $\fp$.

\begin{remark}[CLR/Bargmann as an unconditional tool]
The number of negative eigenvalues of $\kin-\vv$ is bounded by functionals of the well
(Cwikel--Lieb--Rozenblum, Bargmann, Lieb--Thirring) that depend \emph{only} on
$\vv_\lambda$ (prime side, PNT), \emph{not} on the zeros. A bound
$N_-(\kin-\vv_\lambda)\le1$ would be unconditional. \textbf{Internal consistency:}
$A_\lambda=(\kin-\vv)+\pole\succeq0$ with spectrum $(k+1)^2\varepsilon_0>0$ (Weyl
tower, \NM) implies that $\kin-\vv=A_\lambda-\pole$ has at most \emph{one} negative
eigenvalue (rank-$1$ subtraction), aligned with $\fp$ --- exactly the picture the
Conjecture predicts.
\end{remark}

\paragraph{The residual \GAP, localized.}
What is missing to \PR\ Conjecture \ref{conj:hardy} / \GAP\ (i):
\begin{enumerate}[label=(\alph*),leftmargin=2em]
\item \GAP\ \textbf{Non--locality.} $\kin$ is not the local $-d^2/dx^2$ but a jump
generator (Lévy). Classical Hardy improvements and CLR bounds are for standard local
or fractional operators; here their analogue is needed for the specific symbol
$\Omega$. (Non--locality --- $85\%$ of the weight in large jumps --- was the
historical obstacle; here it reappears, but \emph{isolated} to this inequality.)
\item \GAP\ \textbf{Alignment of the extremal.} \PR\ that the Hardy extremal
is $\fp$ (currently \NM\ via route \#9). Candidate: the extremal is the eigenvector of
smallest eigenvalue of $\kin-\vv$, and route \#9 identified it with the uniform
direction.
\end{enumerate}

\section{Status and next step}\label{sec:status}
% =====================================================================

\begin{enumerate}[leftmargin=1.6em]
\item \PR\ (Lemma \ref{lem:schur}) Schur reduction: $A_\lambda\succeq0\Leftrightarrow$
(i)~$\kin-\vv\succeq0$ on $\fp^\perp$ $+$ (ii)~scalar inequality. Exact, without
hypotheses.
\item \PR/\NM\ (Prop.\ \ref{prop:extremal}) The Hardy deficit is the extremal
$=\fp$; route \#9 confirms it (rank $1$, pole direction).
\item \OP\ (Conjecture \ref{conj:hardy}) Improved non--local Hardy inequality:
$\kin-\vv\succeq0$ on $\fp^\perp$, with remainder. \emph{Unconditional} (only $K,V$, no
zeros). Reduced to \GAP\ (a)~non--locality and (b)~alignment of the extremal.
\item \textbf{Consequence if (i) goes through:} the whole problem collapses to the
single scalar Schur inequality (ii), $b+\kappa_\lambda\ge r^*M^+r$ --- a single
numerical comparison between quantities $\sim\lambda^{-2}$, both computable on the
prime side.
\end{enumerate}

\noindent\textbf{Honesty.} \GAP\ (i) has been reduced, with exact algebra
(Schur) $+$ a classical mechanism (Hardy improvement upon projecting out the
extremal), to an \emph{unconditional} functional inequality whose only obstacle is the
non--locality of the archimedean symbol. This is \textbf{not} RH in disguise: it is a
spectral-theory inequality (bound-state counting for a non--local Schrödinger
operator) on prime-side objects. The battle for (i) is now the battle for a
non--local Hardy improvement --- classical ground, with tools (CLR, Lieb--Thirring,
Brezis--Vázquez) and a concrete obstacle.
