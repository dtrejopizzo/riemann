\begin{abstract}
The question that would decide whether the CCM program gives a \emph{non-circular}
bridge to RH was: does the lower bound $\varepsilon_0(\lambda)\ge c/\lambda^2$ follow
from the \emph{density} of the zeros (PNT/Landau, non-circular) or from their
\emph{position} (RH)? The answer, once two operators the program had been conflating
are carefully separated, is sharp: \textbf{the $\varepsilon_0$ proved unconditionally
is the smallest eigenvalue of the \emph{frame} operator $\Sframe$ (pure density,
Landau--Beurling), and RH is \emph{not} that object: RH is the coincidence
$\Aweil=\Sframe$ (the Weil form equals the frame), which is exactly the
\emph{position} condition.} The program delivers genuine, unconditional density
results; RH remains \emph{outside} its non-circular reach, in the gap
$\Aweil-\Sframe$, which is RH by definition (Weil's criterion). Verdict: a clean
reformulation, not a bridge.
\end{abstract}

\section{Two operators that must be separated}\label{sec:two}
% =====================================================================

Let $\El=\mathrm{PW}_{L/2}$, $L=2\log\lambda$, and $\{\gamma_j\}$ the heights of the
non-trivial zeros $\rho_j=\beta_j+i\gamma_j$ of $\zeta$ with $\gamma_j$ below the
horizon $T^*\sim2\pi\lambda^2$.

\begin{definition}[Frame operator $\Sframe$ --- density]
\[
  \Sframe(f,f)\;:=\;\sum_{j}\bigl|\widehat f(\half+i\gamma_j)\bigr|^2,
  \qquad
  \varepsilon_0^{\Sframe}(\lambda):=\min_{f\in\El,\;\|f\|=1}\Sframe(f,f).
\]
This is the sampling form of $\widehat f$ at the \emph{real heights} $\gamma_j$.
It is $\ge0$ by construction (sum of squares); $>0$ iff $\{\gamma_j\}$ is a
\emph{frame} for $\El$, which depends \emph{only on the density} of the
$\gamma_j$ (Landau/Beurling density theorem).
\end{definition}

\begin{definition}[Weil form $\Aweil$ --- position]
\[
  \Aweil(f,f)\;:=\;\sum_{j}\widehat f(\rho_j)\,\overline{\widehat f(1-\overline{\rho_j})}
  \;=\;\sum_j \widehat f(\beta_j+i\gamma_j)\,\overline{\widehat f(1-\beta_j+i\gamma_j)} .
\]
This is the form that appears in the explicit formula; it involves the \emph{full
positions} $\rho_j$ (real part $\beta_j$ included).
\end{definition}

% =====================================================================
\section{The key identity: $\Aweil=\Sframe \Leftrightarrow$ RH}\label{sec:key}
% =====================================================================

\begin{lemma}[Frame--Weil coincidence]\label{lem:coincide}
For an individual zero $\rho_j$:
\[
  \widehat f(\rho_j)\,\overline{\widehat f(1-\overline{\rho_j})}
  \;=\;\bigl|\widehat f(\half+i\gamma_j)\bigr|^2
  \quad\Longleftrightarrow\quad
  \beta_j=\half .
\]
Indeed, $\rho_j=\half+i\gamma_j$ gives $1-\overline{\rho_j}=\half+i\gamma_j=\rho_j$,
and the term becomes $|\widehat f(\rho_j)|^2$. Conversely, if
$\beta_j\ne\half$ then $\rho_j\ne1-\overline{\rho_j}$ and the term is a cross
product $\widehat f(\rho_j)\overline{\widehat f(1-\overline{\rho_j})}$, in general
\emph{not} real $\ge0$.
\end{lemma}

\begin{theorem}[The program's $\varepsilon_0$ is density; RH is position]
\label{thm:verdict}
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=2em]
\item \PR\ \emph{(Density, unconditional.)} $\varepsilon_0^{\Sframe}(\lambda)>0$ for
all $\lambda$, with $\varepsilon_0^{\Sframe}(\lambda)\sim C_0/\lambda^2$ (critical
sampling deficit, Landau--Beurling). \emph{This is what the program measured and
``proved''}; it does not use the position of the zeros, only their density
$\tfrac1{2\pi}\log\tfrac{T}{2\pi}$.
\item $\Aweil(f,f)=\Sframe(f,f)$ for all $f\in\El$ and all $\lambda$
$\;\Longleftrightarrow\;$ $\beta_j=\half$ for all $\gamma_j<T^*(\lambda)$
$\;\Longleftrightarrow\;$ RH below the horizon.
\item \OP\ \emph{(Position $=$ RH.)} $\Aweil(f,f)\ge0$ for all $f\in\El$
$\;\Longleftrightarrow\;$ RH below the horizon. The lower bound
$\varepsilon_0^{\Aweil}(\lambda)\ge c/\lambda^2$ is therefore RH-equivalent.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) is Landau's density theorem applied to the frame $\{\gamma_j\}$ at critical
density $L/2\pi$; the rate $C_0/\lambda^2$ is the deficit at the edge
$T^*\sim2\pi\lambda^2$. (b) is Lemma \ref{lem:coincide} summed over $j$ (the
terms coincide one by one iff each $\beta_j=\half$). (c): if every $\beta_j=\half$
then $\Aweil=\Sframe\ge0$; if some $\beta_{j_0}\ne\half$ with $\gamma_{j_0}<T^*$,
choose $f$ spectrally concentrated near $\rho_{j_0}$ and $1-\overline{\rho_{j_0}}$
with opposite phases, making $\Aweil(f,f)<0$; by homogeneity
$\varepsilon_0^{\Aweil}\le0$. \qed
\end{proof}

\section{Why the confusion mattered}\label{sec:confusion}
% =====================================================================

\begin{remark}[The source of the mirage]
Throughout the program, $\varepsilon_0$ was treated as ``the Weil-positivity
margin'' \emph{and} as ``the smallest eigenvalue of the frame of the zeros'', as if
they were the same object. Theorem \ref{thm:verdict} shows that they coincide
\emph{exactly under RH} (Lemma \ref{lem:coincide}) and \emph{only} then. Since all
the numerics use the \emph{true} zeros (which are on the line up to enormous
heights), what was always measured was the regime $\Aweil=\Sframe$, and the observed
$\varepsilon_0\sim C_0/\lambda^2$ is that of the \emph{frame} (density). The RH
content --- the possibility $\Aweil\ne\Sframe$ --- is \emph{invisible} to that
measurement, because it would require a zero off the line, which does not exist in
the computed range.
\end{remark}

\begin{corollary}[What density gives and what it does not]
\begin{itemize}[leftmargin=1.4em]
\item \PR\ Density (PNT/Landau) gives $\varepsilon_0^{\Sframe}\sim C_0/\lambda^2>0$:
real, unconditional, but it is a \emph{sampling} property, not RH.
\item The classical zero-free regions (Vinogradov--Korobov) give a \emph{partial}
and \emph{non-uniform} bound on $\beta_j$ ($\beta_j<1-c/(\log\gamma_j)^{2/3}$),
which controls $\Aweil-\Sframe$ \emph{only} in a narrowing strip; \textbf{it does not}
reach $\beta_j=\half$.
\item \OP\ The jump from ``narrowing strip'' to uniform $\beta_j=\half$ is exactly
RH. No density input covers it.
\end{itemize}
\end{corollary}

\section{Program verdict}\label{sec:verdict}
% =====================================================================

\begin{enumerate}[leftmargin=1.6em]
\item \textbf{Density or position:} \textbf{position}. The bound
$\varepsilon_0\ge c/\lambda^2$ \emph{relevant to RH} is that of the Weil operator
$\Aweil$, and $\Aweil\ge0\Leftrightarrow$ RH. The frame bound $\Sframe$ (density) is
unconditional but \emph{is not} RH.
\item \textbf{What the program DID achieve} (unconditional, real):
\begin{itemize}[leftmargin=1.3em]
\item \PR\ Link 1 (finite real-rootedness, H1+H2): truncated $\xi$'s with only
real zeros, without assuming RH.
\item \PR\ The faithful CCM engine (zeros to $10^{-50}$ with few primes).
\item \PR\ $\varepsilon_0^{\Sframe}\sim C_0/\lambda^2$ (critical density,
Landau--Beurling): a clean sampling theorem.
\item \PR\ The anatomy of the margin: tower $(k+1)^2$, rank-1 $\mathcal L$,
pole-direction --- frame structure, not RH structure.
\end{itemize}
\item \textbf{What remains outside non-circular reach:} RH itself. The last step
($\Aweil\ge0$, or $\Aweil=\Sframe$) is the Weil-positivity criterion, RH by
definition. \emph{The program is a clean reformulation and a source of
unconditional density results, but not a non-circular bridge to RH.}
\end{enumerate}

\noindent\textbf{Final honesty.} This is the expectable outcome of any attack on RH
via Weil positivity: the criterion is RH-equivalent, and separating density from
position shows that the provable part (density/frame) does not touch the RH part
(position). The value of the program is not in closing RH --- it does not close it
--- but in (i)~the unconditional results of Links 0, 1 and of the frame, and
(ii)~having located, with proof, that the wall is $\Aweil-\Sframe=$ position
$=$ RH, and not an operator $\mathcal L$ yet to be discovered nor an Euler tail
yet to be bounded.
