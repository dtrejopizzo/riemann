\begin{abstract}
Granted that Link 2 (convergence $\xih_{\lambda}\to\Xi$) reduces, once the
boundary-layer issue is settled, to controlling the \emph{prime-truncation floor}
$\mathrm{piso}_\lambda(n)$, this document attacks that object with the explicit
formula. The central result is \emph{negative and precise}:
\textbf{via the standard route (Perron + explicit formula), the bound on the floor is
RH--equivalent} (Theorem \ref{thm:circ}). The floor is expressed intrinsically as
a sum over the zeros $\rho$, and its optimal size (square root) \emph{is}
RH. This identifies Scenario C rigorously for the naive route, and isolates
the \emph{only} escape route (Scenario A): the archimedean$\leftrightarrow$primes
cross-cancellation of the combined symbol $W_\lambda$
(\S\ref{sec:escape}), which is the localized open problem.
\end{abstract}

% =====================================================================
\section{The floor as a zero shift}\label{sec:shift}
% =====================================================================

Let $\Xi$ be Riemann's completed $\xi$ and $\xih_\lambda$ the entire function of the
CCM model built with primes $p\le\lambda^2$ (cf.\ the main document). In the
regime where the band is saturated ($N\to\infty$ taken, type $L/2=\log\lambda$
fixed by $\lambda$), the difference of the logarithmic symbols is
\begin{equation}\label{eq:Rlam}
  R_\lambda(s)\;:=\;-\frac{\zeta'}{\zeta}(s)\;-\;\sum_{n\le\lambda^2}\frac{\Lam(n)}{n^{s}},
\end{equation}
the \emph{remainder of the truncated von Mangoldt sum}. The first summand of
\eqref{eq:Rlam} is the full series (analytic continuation), the second a
Dirichlet polynomial (entire).

\begin{proposition}[Shift of a zero]\label{prop:shift}
Let $\rho_n=\half+i\gamma_n$ be a simple zero of $\Xi$ and $\widehat\rho_n(\lambda)$ the
corresponding zero of $\xih_\lambda$. To first order in the perturbation of the
symbol,
\begin{equation}\label{eq:shiftformula}
  \widehat\rho_n(\lambda)-\rho_n
  \;=\;-\,\frac{R_\lambda(\rho_n)}{\Phi'(\rho_n)}\;\bigl(1+o(1)\bigr),
\end{equation}
where $\Phi(s)=\frac{d}{ds}\log\Xi(s)$ is the logarithmic density
(with $\Phi'(\rho_n)\ne0$ by simplicity). In particular
\[
  \mathrm{piso}_\lambda(n)\;=\;|\widehat\rho_n(\lambda)-\rho_n|
  \;\asymp\;|R_\lambda(\half+i\gamma_n)| .
\]
\end{proposition}
\begin{proof}[Idea]
The zero of $\xih_\lambda$ is the zero of the combined symbol perturbed by
$R_\lambda$; by the implicit function theorem applied to the characteristic
equation $\Phi(s)+R_\lambda(s)=0$ near $\rho_n$, the shift is
$-R_\lambda(\rho_n)/\Phi'(\rho_n)$. \qed
\end{proof}

\noindent\textbf{Conclusion of \S\ref{sec:shift}.} The floor is, up to the
normalization factor $1/\Phi'(\rho_n)$ (universal, independent of $\lambda$), the
\emph{value of the truncated remainder evaluated on the critical line at the
height of the zero}. Everything hinges on $R_\lambda(\half+it)$.

% =====================================================================
\section{The truncated remainder via the explicit formula}\label{sec:explicit}
% =====================================================================

We compute $R_\lambda$ by Abel summation starting from the Chebyshev function
$\psi(u)=\sum_{n\le u}\Lam(n)$ and its explicit formula.

\begin{lemma}[Representation of the remainder]\label{lem:rep}
For $\Real(s)\in(\half,1)$ and $y=\lambda^2$,
\begin{equation}\label{eq:rep}
  \sum_{n\le y}\frac{\Lam(n)}{n^{s}}
  \;=\;\frac{s}{s-1}\,\bigl(1-y^{1-s}\bigr)
  \;-\;s\sum_{\rho}\frac{y^{\rho-s}}{\rho(\rho-s)}
  \;+\;E(s,y)\;+\;\psi(y)\,y^{-s},
\end{equation}
where the sum runs over the nontrivial zeros $\rho$, and $E(s,y)$ collects the
smooth archimedean terms (trivial pole, $\zeta'/\zeta(0)$), uniformly
$O(1)$. Taking $y\to\infty$ with $\Real s>1$ recovers $-\zeta'/\zeta(s)$, and
hence
\begin{equation}\label{eq:Rrep}
  R_\lambda(s)\;=\;\underbrace{\frac{s}{s-1}\,y^{1-s}}_{\text{pole at }s=1}
  \;+\;s\sum_{\rho}\frac{y^{\rho-s}}{\rho(\rho-s)}
  \;-\;\psi(y)\,y^{-s}\;-\;E(s,y),
  \qquad y=\lambda^2 .
\end{equation}
\end{lemma}
\begin{proof}
Abel summation:
$\sum_{n\le y}\Lam(n)n^{-s}=\psi(y)y^{-s}+s\int_1^y\psi(u)u^{-s-1}\,du$.
Inserting $\psi(u)=u-\sum_\rho u^\rho/\rho-\log(2\pi)-\half\log(1-u^{-2})$
(Riemann--von Mangoldt explicit formula) and integrating term by term:
$s\int_1^y u\cdot u^{-s-1}du=\frac{s}{1-s}(y^{1-s}-1)$ and
$-s\int_1^y\frac{u^\rho}{\rho}u^{-s-1}du=-\frac{s}{\rho(\rho-s)}(y^{\rho-s}-1)$;
the remaining terms are $E(s,y)=O(1)$. Rearranging gives \eqref{eq:rep}. \qed
\end{proof}

\begin{remark}[The signature of Scenario C, explicit]\label{rem:C}
The oscillatory part of $R_\lambda$ \emph{is, literally, a sum over the zeros}:
\[
  R_\lambda^{\mathrm{osc}}(s)\;=\;s\sum_\rho\frac{(\lambda^2)^{\rho-s}}{\rho(\rho-s)} .
\]
The dependence on the \emph{position} of the zeros is not an artifact: it appears
structurally in the only available representation of the truncated remainder.
This is the danger of Scenario C, made concrete.
\end{remark}

% =====================================================================
\section{Theorem: the standard route is RH--equivalent}\label{sec:circ}
% =====================================================================

On the critical line $s=\half+it$ with $t=\gamma_n$, the modulus of each oscillatory
term of \eqref{eq:Rrep} is
\[
  \left|(\lambda^2)^{\rho-s}\right|
  \;=\;(\lambda^2)^{\,\Real\rho-\half}
  \;=\;(\lambda^2)^{\,\beta-\half},
  \qquad \beta:=\Real\rho .
\]

\begin{theorem}[Circularity of the standard bound]\label{thm:circ}
Let $\Theta:=\sup_\rho\Real\rho$ be the supremum of the real parts of the zeros.
Then the bound on the floor obtained from representation \eqref{eq:Rrep} satisfies
\begin{equation}\label{eq:bound}
  \mathrm{piso}_\lambda(n)\;\asymp\;|R_\lambda(\half+i\gamma_n)|
  \;\lesssim\;(\lambda^2)^{\,\Theta-\half}\cdot\Psi(\gamma_n),
\end{equation}
with $\Psi(\gamma_n)=\sum_\rho\frac{1}{|\rho|\,|\rho-\half-i\gamma_n|}$ convergent.
Moreover the bound is optimal in the exponent. Consequently:
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=2em]
\item \emph{Unconditionally} only $\Theta\le1$ is known, giving
$\mathrm{piso}_\lambda\lesssim(\lambda^2)^{1/2}=\lambda$ --- a \textbf{useless bound}
(it does not decay).
\item The optimal square-root bound $(\lambda^2)^{\Theta-1/2}=O(1)$ uniform,
i.e.\ a bounded floor subsequently decreasing via the subdominant factors,
\textbf{is equivalent to} $\Theta=\half$, \textbf{which is exactly RH}.
\end{enumerate}
\end{theorem}
\begin{proof}
The triangle inequality in \eqref{eq:Rrep} gives
$|R_\lambda^{\mathrm{osc}}|\le|s|(\lambda^2)^{\Theta-1/2}\sum_\rho|\rho|^{-1}|\rho-s|^{-1}$;
the pole $\frac{s}{s-1}(\lambda^2)^{1-s}$ contributes $(\lambda^2)^{1/2}$ which is dominated
or cancelled by the archimedean counter-term (see \S\ref{sec:escape}); the
term $\psi(y)y^{-s}$ contributes $\sim y^{1/2}/y^{1/2}=O(1)$ by PNT. The convergence
of $\Psi$ follows from $\sum_\rho|\rho|^{-2}<\infty$ and from $\gamma_n$ being at
positive distance from the other zeros. The optimality of the exponent $\Theta-\half$:
there exists $\rho^\ast$ with $\Real\rho^\ast$ arbitrarily close to $\Theta$, and its
term cannot be cancelled by the others for generic $\lambda$ (equidistributed phases
$(\lambda^2)^{i\,\Im\rho}$), so the $\limsup$ realizes the
exponent. The equivalence with RH in (b) is the definition $\Theta=\half
\Leftrightarrow$ all $\beta=\half$. \qed
\end{proof}

\begin{corollary}[The wall, sharply stated]
Proving $\mathrm{piso}_\lambda(n)\to0$ at rate $(\lambda^2)^{-\delta}$, $\delta>0$,
via representation \eqref{eq:Rrep} \emph{requires} $\Theta\le1-\delta'$, i.e.\
a quantitative zero-free region; and the optimal square-root rate is
$\Theta=\half=$ RH. The standard route \textbf{cannot} close Link 2 without
assuming (part of) what it wants to prove.
\end{corollary}

\noindent This confirms, now with a proof, what the numerics only
hinted at and what had already appeared as the ``adjusted constant $c=2$'' of
Target A: \textbf{the prime remainder, treated in isolation, is RH--circular}.

% =====================================================================
\section{The only escape route: cross-cancellation}\label{sec:escape}
% =====================================================================

Scenario A is not refuted: what Theorem \ref{thm:circ} refutes is
\emph{treating the prime remainder in isolation}. The CCM model does not use $R_\lambda$
alone, but the \emph{combined symbol}
\begin{equation}\label{eq:Wlam}
  W_\lambda(t)\;=\;\underbrace{2\theta'(t)}_{\text{archimedean}}
  \;-\;2\,\Real\!\Bigl[\textstyle\sum_{n\le\lambda^2}\frac{\Lam(n)}{n^{1/2+it}}\Bigr],
  \qquad
  2\theta'(t)\sim\log\frac{t}{2\pi},
\end{equation}
and the Weil form $A_\lambda(g,g)=\frac1{2\pi}\int|\widehat g|^2 W_\lambda\,dt$.

\begin{remark}[Why the combined object can escape]\label{rem:escape}
The archimedean term $2\theta'(t)$ and the prime sum come from the
\emph{two sides} of the explicit formula: for a test function $g$,
\[
  \sum_\rho \widehat g(\gamma_\rho)
  \;=\;\underbrace{\frac1{2\pi}\!\int \widehat g(t)\,2\theta'(t)\,dt}_{\text{archimedean}}
  \;-\;\underbrace{2\sum_n\frac{\Lam(n)}{\sqrt n}\,\check g(\log n)}_{\text{primes}} .
\]
The oscillatory part of \emph{both} sides is the same sum $\sum_\rho$; when
combined into $W_\lambda$, the zero-representation of
\S\ref{sec:explicit} \textbf{cancels against the archimedean term}. That is: the
$\sum_\rho$ that makes $R_\lambda$ circular \emph{does not appear} in the correct
combined object --- it appears as the \emph{difference} between the truncated and the
full one, which is a smooth Fourier tail.
\end{remark}

\begin{proposition}[Non-circular reformulation of the floor]\label{prop:noncirc}
Define the truncated combined remainder
\[
  \mathcal{W}_\lambda(t)\;:=\;W_\infty(t)-W_\lambda(t)
  \;=\;2\,\Real\!\Bigl[\sum_{n>\lambda^2}\frac{\Lam(n)}{n^{1/2+it}}\Bigr]_{\text{(cancelled sum)}} .
\]
Then the floor is $\mathrm{piso}_\lambda(n)\asymp|\mathcal W_\lambda$ weighted by
$|\widehat g_n|^2|$, and Scenario A is equivalent to the bound
\begin{equation}\label{eq:Atarget}
  \boxed{\;\Bigl|\frac1{2\pi}\!\int |\widehat g_n(t)|^2\,\mathcal W_\lambda(t)\,dt\Bigr|
  \;\le\; C\cdot\eta(\lambda),\qquad \eta(\lambda)\to0,\;}
\end{equation}
with $\eta$ \emph{controllable by PNT} (prime distribution, classical zero-free
regions of Vinogradov--Korobov type), \textbf{without} using $\Theta=\half$.
\end{proposition}

\paragraph{The \GAP, pinpointed exactly.}
The quantity $\mathcal W_\lambda(t)=2\Real\sum_{n>\lambda^2}\Lam(n)n^{-1/2-it}$ is,
crudely, a divergent tail; \emph{cancelled} (summed via the phase $n^{-it}$),
it is the Fourier tail of the full symbol. The open question is:
\begin{quote}\itshape
Does a bound \eqref{eq:Atarget} exist for the tail \emph{weighted by the
band-limited window} $|\widehat g_n|^2$ (supported on $|\log n|\le\log\lambda$), using
only classical PNT/zero-free regions and \emph{not} the hypothesis
$\Theta=\half$?
\end{quote}
The window $|\widehat g_n|^2$ is band-limited of type $\log\lambda$: it \emph{only sees}
$\log n\le\log\lambda$, i.e.\ $n\le\lambda$, while the tail starts at $n>\lambda^2$.
\textbf{There is a spectral gap} (between $\lambda$ and $\lambda^2$) that is exactly the
``room'' where the cross-cancellation could give the bound without zeros. This is the
working hypothesis of the Connes program (cancellation between places /
$\mathbb{S}$-local terms), here reduced to a concrete analytic inequality.

\section{Verdict}\label{sec:verdict}
% =====================================================================

\begin{enumerate}[leftmargin=1.6em]
\item \PR{} (Theorem \ref{thm:circ}) Treating the prime remainder $R_\lambda$ in
isolation gives an \textbf{RH--equivalent} bound on the floor: $\Theta=\half\Leftrightarrow$
optimal rate. The naive route for Link 2 is circular. \emph{This is a
result, not a conjecture.}
\item \OP{} (Prop.\ \ref{prop:noncirc}) The only non-circular route is the bound
\eqref{eq:Atarget} on the combined symbol $\mathcal W_\lambda$, exploiting the spectral
gap $[\lambda,\lambda^2]$ and the archimedean$\leftrightarrow$primes cancellation.
It is a concrete analytic problem, not an operator identification.
\item \textbf{Position on the map.} Link 2 does not close via elementary
analysis; but the obstacle has been \emph{proved} circular on the standard route
and \emph{isolated} into a single inequality on the Connes route. The candidate
tools (Selberg--Beurling method for the window, Vinogradov-type sums for the tail,
localized Weil positivity for the weight) are classical analytic number theory.
\end{enumerate}

\noindent\emph{Honesty:} this document does \textbf{not} prove \eqref{eq:Atarget};
it proves that \eqref{eq:Atarget} is the only thing missing, and that its negation via the
standard route is RH--circular. The next block of work is to attack
\eqref{eq:Atarget} with the spectral gap $[\lambda,\lambda^2]$ as the central resource.
