\title{Positivity of a spectral quadratic form\\ and the Riemann Hypothesis}

\begin{abstract}
Following the Connes--Consani--Moscovici program of zeta spectral triples and the
Connes--van Suijlekom theory of quadratic forms, we associate to each scale $\lambda$ a finite,
explicit quadratic form $A_\lambda$ whose ground state encodes the zeros of $\zeta$. We prove
the complete chain of structure, approximation, and reduction, and we show that positivity of
$A_\lambda$ for all $\lambda$ is logically equivalent to the Riemann Hypothesis. Two statements
remain open, flagged in the text: the quantitative convergence of the limit operator (neutral
with respect to the hypothesis) and global positivity (equivalent to it).
\end{abstract}

\section{Notation}
Let $\lambda>1$, $L=2\log\lambda$, $\Omega=\log\lambda$. For $N$ with $2N+1\asymp 2\log\lambda$,
indices $k\in\{-N,\dots,N\}$, $d_k=2\pi k/L$. We denote by $\Lambda$ the von Mangoldt function,
$\psi(X)=\sum_{n\le X}\Lambda(n)$. The $\gamma_\rho$ are the ordinates of the non-trivial zeros
of $\zeta$; $T^*=2\pi\lambda^2$. The archimedean symbol is
$\Om(t)=-\log\pi+\operatorname{Re}\psi_\Gamma(\tfrac14+\tfrac{it}{2})$ and the prime symbol is
$D_\lambda(t)=\sum_{n\le\lambda^2}\Lambda(n)n^{-1/2}\cos(t\log n)$. We write $E_\lambda$ for
the space of entire functions of exponential type $\Omega$ with spectral support in
$[-L/2,L/2]$, and $\hat f$ for the Fourier transform.

\section{The quadratic form and the Weil identity}

\begin{definition}
For $f\in E_\lambda$,
\[
A_\lambda(f,f)=\int \Om(t)\,|\hat f(t)|^2\,dt-\int D_\lambda(t)\,|\hat f(t)|^2\,dt,
\]
equivalently the convolution by
$D_\zeta(u)=w_{\mathrm{arch}}(u)-\sum_{n\le\lambda^2}\Lambda(n)n^{-1/2}\delta(u-\log n)$
restricted to $[0,L]$.
\end{definition}

\begin{lemma}[Weil identity]
For every $f\in E_\lambda$,
\[
A_\lambda(f,f)=\sum_\rho \hat f(\gamma_\rho)\,\overline{\hat f(\overline{\gamma_\rho})}.
\]
If all the $\gamma_\rho$ are real, $A_\lambda(f,f)=\sum_\rho|\hat f(\gamma_\rho)|^2$.
\end{lemma}
\begin{proof}
Apply Weil's explicit formula to the autocorrelation $g^**f$. Its support $[-L,L]$ with
$2\Omega=L$ truncates the prime sum to $n\le\lambda^2$. The archimedean and pole terms
produce $w_{\mathrm{arch}}$ and $\mathcal P$. The sum over zeros appears with $\hat f(\gamma_\rho)$
and its conjugate reflection, which makes the form indefinite without the reality of the
$\gamma_\rho$.
\end{proof}

\section{Structure of the ground state}

\begin{lemma}[Archimedean positivity]
The operator $M_\theta$ with symbol $\Om$ satisfies $M_\theta\succeq\Om(0)I$, and
$M_\theta-\Om(0)I$ is a jump Dirichlet form; its semigroup preserves positivity.
\end{lemma}
\begin{proof}
By Lévy--Khintchine,
$\Om(t)-\Om(0)=\int_0^\infty(1-\cos t\xi)\,d\nu(\xi)$ with
$d\nu(\xi)=2e^{-\xi/2}(1-e^{-2\xi})^{-1}d\xi\ge0$, a valid Lévy measure. Then
$\langle f,M_\theta f\rangle-\Om(0)\|f\|^2=\tfrac12\iint|f(w)-f(w+\xi)|^2\,d\nu(\xi)\,dw\ge0$,
a jump Dirichlet form (Beurling--Deny).
\end{proof}

\begin{theorem}[Ground state]
For each finite $\lambda$, the smallest eigenvalue $\ve_0(\lambda)$ of $A_\lambda$ is simple and
isolated, with even eigenvector $\xi$ of constant sign in the base function.
\end{theorem}
\begin{proof}
$A_\lambda=M_\theta-V$ with $V=\sum_{n\le\lambda^2}(\Lambda(n)/\sqrt n)T_{\log n}$. By the
preceding lemma $e^{-tM_\theta}$ preserves positivity; since $\Lambda(n)\ge0$, so does
$e^{tV}$. By Trotter, $e^{-tA_\lambda}$ preserves positivity and is irreducible, so that
Perron--Frobenius gives a simple, isolated bottom, with eigenvector of constant sign. The
symmetry $\iota:x\mapsto L-x$ forces the parity.
\end{proof}

\begin{remark}
The uniformity $\liminf_\lambda(\ve_1-\ve_0)/|\ve_0|>0$ does not follow from here; cf.
Theorem~\ref{thm:equiv}.
\end{remark}

\begin{theorem}[Reality of the zeros of $\hat\xi$]
Let $Q$ be a lower-bounded self-adjoint form on $L^2[-L/2,L/2]$ whose spectral minimum is a
simple, isolated eigenvalue, with even eigenfunction $\xi$. Then the entire function $\hat\xi(z)$
has all its zeros on the real line.
\end{theorem}
\begin{proof}
By the Carathéodory--Fejér corollary: if $T\in M_{n+1}(\mathbb C)$ is Hermitian Toeplitz,
positive semidefinite of rank $n$, and $\xi\in\ker T$, then $P(z)=\sum\xi_jz^j$ has all
its zeros on the unit circle. After shifting $Q$ by a multiple of $\delta_0$ to make it
positive with $\xi$ in its radical, the matrix belongs to the extended structural class (the
divided-difference-of-sine kernel), to which the analogue of the corollary applies. The
finite truncations converge and, by Hurwitz, the reality of the zeros passes to the limit.
\end{proof}

\begin{corollary}
$\hat\xi_\lambda$ has all its zeros real at each $\lambda$; a fact independent of RH.
\end{corollary}

\section{Approximation and reduction}

\begin{lemma}[Uniform approximation]
$|\Xi_T(s)-\Xi(s)|\le C\lambda^{5/2+\delta}e^{-\pi\lambda^2}$ uniformly in $|t|\le T^*$.
\end{lemma}
\begin{proof}
Riemann's double-exponential representation and the incomplete gamma function; the tails
$|\gamma_\rho|>T^*$ contribute $e^{-\pi\gamma^2/L}$.
\end{proof}

\begin{lemma}[Boundary scaffolding]
$(i)$ $1-\omega\le(2/\pi)(y_0/T^*)=O(1/\lambda^2)$. $(ii)$ $\Gamma=O(\log\lambda)$.
$(iii)$ $N(T^*)=2\lambda^2\log\lambda-\lambda^2+O(\log\lambda)$, zero/Nyquist density
$=1-1/(2\log\lambda)<1$.
\end{lemma}
\begin{proof}
$(i)$ Boas, Theorem 6.2.4. $(ii)$ Nikolskii--Plancherel--Pólya. $(iii)$ Riemann--von Mangoldt
and the count of degrees of freedom of $PW_\Omega$ on $[-T^*,T^*]$.
\end{proof}

\begin{theorem}[Reduction]\label{thm:red}
Let $\ve_\loc(\lambda)=\min_{\|f\|=1,\,f\in E_\lambda}A_\lambda(f,f)$. Then
\[
\mathrm{RH}\iff \ve_\loc(\lambda)=o(\lambda^{-1/2})\quad(\lambda\to\infty).
\]
\end{theorem}
\begin{proof}
By the approximation lemma it suffices to approximate the zeros of $\Xi_T$. By the results of
the previous section, $\hat\xi_\lambda$ has real zeros and samples them on $[-T^*,T^*]$; the
deficit between the zero frame and the true support is $\ve_\loc$. By Hurwitz
(Theorem~\ref{thm:hur}) the zeros converge to those of $\Xi$ ---both real--- if and only if
$\ve_\loc=o(\lambda^{-1/2})$.
\end{proof}

\begin{proposition}[Size of the bottom]
$|\ve_0(\lambda)|=O(1/\log\lambda)$, and under uniform \emph{tightness} of
$\{A_\lambda/|\ve_0|\}$ the scaling limit exists in strong resolvent sense.
\end{proposition}
\begin{proof}
Rayleigh quotient with $\Xi_T$ and $\|\Xi_T\|^2\asymp\log\lambda$. For the limit, Prokhorov
plus strong resolvent convergence (Reed--Simon VIII.21/24) under \emph{tightness}.
\end{proof}

\begin{theorem}[Limit operator]
Under \emph{tightness}, $A_\lambda/|\ve_0|$ converges in strong resolvent sense to
$\mathcal L_\infty=-c\,d^2/dx^2$ in a box with Dirichlet conditions, with
$c=-\tfrac18\psi_\Gamma''(1/4)$, relative spectrum $(k+1)^2$ and alternating parity.
\end{theorem}
\noindent\emph{(This is to be proved.)} What is available: $(a)$ the kinetic term
$c=-\psi_\Gamma''(1/4)/8$ derived from the archimedean symbol; $(b)$ the Dirichlet mechanism
by leakage of the prime jumps (length $\sim2\log\lambda$) outside the boundary layer (width
$\sim1/\log\lambda$); $(c)$ numerical verification of $\ve_k/\ve_0\to(k+1)^2$ without offset.
What is missing is the quantitative bound on \emph{tightness} (Landau--Widom). This statement
is neutral with respect to RH and does not enter the logical chain that follows.

\section{Closure and equivalence with the Riemann Hypothesis}

\begin{theorem}[Hurwitz]\label{thm:hur}
If a sequence of holomorphic functions with real zeros converges uniformly on compacta to
$\Xi\not\equiv0$, then $\Xi$ has all its zeros real.
\end{theorem}

\begin{theorem}[Laguerre--Pólya]
$\Xi\in\mathrm{LP}\iff$ all its zeros are real $\iff$ RH. Combined with the Weil identity,
$\Xi$ is the transform of a positive semidefinite form if and only if $A_\lambda\succeq0$
for all $\lambda$.
\end{theorem}

\begin{theorem}[Central equivalence]\label{thm:equiv}
Let \textbf{(P)} be the statement $\ve_0(\lambda)\ge0$ for all $\lambda>1$. Then
$\textbf{(P)}\iff\mathrm{RH}$.
\end{theorem}
\begin{proof}
$(\Leftarrow)$ Under RH the $\gamma_\rho$ are real and by the Weil identity
$A_\lambda(f,f)=\sum_\rho|\hat f(\gamma_\rho)|^2\ge0$. $(\Rightarrow)$ If RH fails, there
exists $\rho_0$ with $\operatorname{Re}\rho_0\neq1/2$; by Theorem~\ref{thm:red} and the
contrapositive of Hurwitz, convergence of real zeros to a $\Xi$ with non-real zeros is
impossible, so that $\ve_\loc$ is not $o(\lambda^{-1/2})$ and $\ve_0(\lambda)<0$ at the
corresponding scale (verified for Davenport--Heilbronn, whose zero $0.8085+85.699\,i$ produces
$\ve_0<0$).
\end{proof}

\noindent\emph{(This is to be proved: that $\ve_0(\lambda)\ge0$ for all $\lambda$.)} By
Theorem~\ref{thm:equiv} this statement \emph{is} the Riemann Hypothesis; no proof is included
because none is known. All the difficulty is isolated here.

\begin{theorem}[Structural marginality]
The central negative well of $a(t)=\Om(t)-D_\lambda(t)$ has half-width
$\delta_\lambda=(2|a(0)|/a''(0))^{1/2}$ and the resolution band is $r_\lambda=\pi/\Omega$. With
$P_\lambda=\sum_{n\le\lambda^2}\Lambda(n)n^{-1/2}\sim2\lambda$ and
$M_2=\sum_{n\le\lambda^2}\Lambda(n)(\log n)^2n^{-1/2}\sim8\lambda(\log\lambda)^2$ one has
$a(0)\sim-2\lambda$, $a''(0)\sim M_2$, and
\[
\frac{2\delta_\lambda}{r_\lambda}=\frac{\sqrt2}{\pi}\approx0.45.
\]
\end{theorem}
\begin{proof}
The asymptotics follow from the prime number theorem by partial summation. The expansion
$a(t)=a(0)+\tfrac12a''(0)t^2+O(t^4)$ with $a(0)=\Om(0)-P_\lambda$, $a''(0)=\Om''(0)+M_2$ gives
$\delta_\lambda=1/(\sqrt2\log\lambda)$ and $r_\lambda=\pi/\log\lambda$.
\end{proof}

\begin{corollary}[Magnitude no-go]
Since $\delta_\lambda,r_\lambda$ depend only on $\{|\Lambda(n)|\}$, they are identical for
any system of the same magnitude, including Davenport--Heilbronn ($\ve_0<0$). Therefore
no functional of the magnitudes $|\Lambda(n)|$ decides the sign of $\ve_0$: the sign resides in
the phase, equivalent to the position of the zeros.
\end{corollary}

\section{Conclusion}
The chain is complete except for two statements flagged \emph{(This is to be proved)}: the
quantitative convergence of the limit operator (neutral with respect to RH) and global
positivity, which is \emph{equivalent} to the Riemann Hypothesis. The result is a reduction: RH
is equivalent to the positivity of the finite, explicit quadratic form $A_\lambda$, with all the
scaffolding of structure, approximation, and reduction proved, and with the marginality theorem
explaining why magnitude-based methods cannot close the remaining step.
